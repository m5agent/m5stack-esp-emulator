+/*
+ * ESP32 flash/cache handoff diagnostic trace
+ *
+ * This is a default-off diagnostic channel. It does not alter controller or
+ * CPU behavior; it records the current virtual time, CPU, guest PC and four
+ * component-specific values in a common line format.
+ */
+
+#ifndef HW_MISC_ESP32_HANDOFF_TRACE_H
+#define HW_MISC_ESP32_HANDOFF_TRACE_H
+
+#include "qemu/osdep.h"
+#include "qemu/timer.h"
+#include "hw/core/cpu.h"
+
+static inline bool esp32_handoff_trace_enabled(void)
+{
+    static int enabled = -1;
+
+    if (enabled < 0) {
+        enabled = getenv("M5SW_ESP32_HANDOFF_TRACE") != NULL;
+    }
+    return enabled;
+}
+
+static inline uint64_t esp32_handoff_trace_sample_interval(void)
+{
+    static uint64_t sample;
+    static bool initialized;
+
+    if (!initialized) {
+        const char *sample_env =
+            getenv("M5SW_ESP32_HANDOFF_TRACE_SAMPLE");
+
+        sample = sample_env ? strtoull(sample_env, NULL, 10) : 1;
+        if (sample == 0) {
+            sample = 1;
+        }
+        initialized = true;
+    }
+    return sample;
+}
+
+static inline void esp32_handoff_trace_event_internal(uint64_t *counter,
+                                                      const char *component,
+                                                      const char *event,
+                                                      uint64_t a, uint64_t b,
+                                                      uint64_t c, uint64_t d,
+                                                      bool always)
+{
+    uint64_t sample;
+    int cpu_index = -1;
+    uint64_t pc = 0;
+
+    if (!esp32_handoff_trace_enabled()) {
+        return;
+    }
+
+    sample = esp32_handoff_trace_sample_interval();
+
+    ++*counter;
+    if (!always && sample > 1 && (*counter % sample) != 0) {
+        return;
+    }
+
+    if (current_cpu) {
+        cpu_index = current_cpu->cpu_index;
+        if (current_cpu->cc && current_cpu->cc->get_pc) {
+            pc = current_cpu->cc->get_pc(current_cpu);
+        }
+    }
+
+    fprintf(stderr,
+            "[esp32-handoff] now=%" PRId64 " cpu=%d pc=%08" PRIx64
+            " component=%s event=%s a=%" PRIu64 " b=%" PRIu64
+            " c=%" PRIu64 " d=%" PRIu64 "\n",
+            qemu_clock_get_ns(QEMU_CLOCK_VIRTUAL), cpu_index, pc,
+            component, event, a, b, c, d);
+    fflush(stderr);
+}
+
+static inline void esp32_handoff_trace_event(uint64_t *counter,
+                                             const char *component,
+                                             const char *event,
+                                             uint64_t a, uint64_t b,
+                                             uint64_t c, uint64_t d)
+{
+    esp32_handoff_trace_event_internal(
+        counter, component, event, a, b, c, d, false
+    );
+}
+
+static inline void esp32_handoff_trace_event_always(uint64_t *counter,
+                                                    const char *component,
+                                                    const char *event,
+                                                    uint64_t a, uint64_t b,
+                                                    uint64_t c, uint64_t d)
+{
+    esp32_handoff_trace_event_internal(
+        counter, component, event, a, b, c, d, true
+    );
+}
+
+static inline void esp32_handoff_trace_pc_sample(uint64_t *counter,
+                                                 uint64_t pc0,
+                                                 uint64_t pc1,
+                                                 unsigned state_flags,
+                                                 uint64_t interval_ns)
+{
+    if (!esp32_handoff_trace_enabled()) {
+        return;
+    }
+
+    ++*counter;
+    fprintf(stderr,
+            "[esp32-handoff] now=%" PRId64
+            " cpu=-1 pc=00000000 component=cpu event=sample"
+            " a=%08" PRIx64 " b=%08" PRIx64
+            " c=%u d=%" PRIu64 "\n",
+            qemu_clock_get_ns(QEMU_CLOCK_VIRTUAL),
+            pc0, pc1, state_flags, interval_ns);
+    fflush(stderr);
+}

+{
+    const char *value =

+static void esp32_handoff_pc_sample_cb(void *opaque)
+{
+    uint64_t pcs[8] = { 0 };
+    unsigned states[8] = { 0 };
+    unsigned state_flags = 0;
+    uint64_t interval_ns = esp32_handoff_pc_sample_interval_ns();
+    CPUState *cpu;
+
+    CPU_FOREACH(cpu) {
+        if (cpu->cpu_index >= 0
+            && cpu->cpu_index < ARRAY_SIZE(pcs)) {
+            if (cpu->cc && cpu->cc->get_pc) {
+                pcs[cpu->cpu_index] = cpu->cc->get_pc(cpu);
+            }
+            states[cpu->cpu_index] =
+                (cpu->halted ? 1 : 0) | (cpu->running ? 2 : 0);
+        }
+    }
+    for (int i = 0; i < 2; ++i) {
+        state_flags |= states[i] << (i * 4);
+    }
+    esp32_handoff_trace_pc_sample(
+        &handoff_pc_sample_count, pcs[0], pcs[1],
+        state_flags, interval_ns
+    );
+    timer_mod(
+        handoff_pc_sample_timer,
+        qemu_clock_get_ns(QEMU_CLOCK_VIRTUAL) + interval_ns
+    );
+}
+
+static void esp32_handoff_pc_sample_start(void)
+{
+    uint64_t interval_ns;
+
+    if (!esp32_handoff_trace_enabled()) {
+        return;
+    }
+    interval_ns = esp32_handoff_pc_sample_interval_ns();
+    if (interval_ns == 0) {
+        return;
+    }
+    if (!handoff_pc_sample_timer) {
+        handoff_pc_sample_timer = timer_new_ns(
+            QEMU_CLOCK_VIRTUAL, esp32_handoff_pc_sample_cb, NULL
+        );
+    }
+    timer_mod(
+        handoff_pc_sample_timer,
+        qemu_clock_get_ns(QEMU_CLOCK_VIRTUAL) + interval_ns
+    );
 }
