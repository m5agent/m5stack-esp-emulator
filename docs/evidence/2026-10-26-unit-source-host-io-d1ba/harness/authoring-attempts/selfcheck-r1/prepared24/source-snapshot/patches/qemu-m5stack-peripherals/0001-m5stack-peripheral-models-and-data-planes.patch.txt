+                                       uint8_t *buffer, uint32_t size)
+{
+    DmaConfigState *state = &s->ch_conf[ESP_GDMA_OUT_IDX][chan];
+    uint32_t addr = ((ESP_GDMA_RAM_ADDR >> 20) << 20)
+                  | FIELD_EX32(state->link, GDMA_OUT_LINK, ADDR);

+    uint32_t consumed = 0;

+     * This is a read-only view of the descriptor ring. Peripheral models use
+     * it to observe data written by the guest without clearing owner bits or
+     * advancing GDMA state.
+     */
+    for (int i = 0; addr && i < 256 && consumed < size; i++) {
+        GdmaLinkedList desc;
+        uint32_t copy;
+

+            break;
+        }
+        copy = MIN(desc.config.length, size - consumed);
+        if (copy && !esp_gdma_read_guest(s, desc.buf_addr,
+                                         buffer + consumed, copy)) {
+            break;
+        }
+        consumed += copy;
+        if (!desc.next_addr || desc.next_addr == start) {
+            break;
+        }
+        addr = desc.next_addr;
+    }
+    return consumed;

+static uint16_t camera_pattern_pixel(M5LcdCamState *s, uint32_t pixel)
+{
+    uint32_t area = MAX(1u, s->width * s->height);
+    uint32_t p = pixel % area;
+    uint32_t x = p % MAX(1u, s->width);
+    uint32_t y = p / MAX(1u, s->width);
+    uint8_t r, g, b;
+
+    if (y < MAX(1u, s->height / 4)) {
+        r = 235; g = 48; b = 48;
+    } else if (y < MAX(1u, s->height / 2)) {
+        r = 235; g = 190; b = 35;
+    } else if (y < MAX(1u, (s->height * 3) / 4)) {
+        r = 40; g = 190; b = 80;
+    } else {
+        r = 35; g = 105; b = 235;
+    }
+    if ((x % 40) < 2 || (y % 40) < 2) {
+        r = g = b = 245;
+    }
+    return ((r >> 3) << 11) | ((g >> 2) << 5) | (b >> 3);
+}

+static void lcd_cam_fill_frame(M5LcdCamState *s, uint8_t *data, uint32_t len)
+{
+    uint32_t frame_bytes = MAX(2u, s->width * s->height * 2);
+    if (s->frame_data && s->frame_data_size >= frame_bytes) {
+        for (uint32_t i = 0; i < len; i++) {
+            data[i] = s->frame_data[(s->frame_offset + i) % frame_bytes];
+        }
+        s->frame_offset = (s->frame_offset + len) % frame_bytes;
+        return;
+    }
+    for (uint32_t i = 0; i < len; i += 2) {
+        uint32_t offset = (s->frame_offset + i) % frame_bytes;
+        uint16_t pixel = camera_pattern_pixel(s, offset / 2);
+        data[i] = pixel & 0xff;
+        if (i + 1 < len) {
+            data[i + 1] = pixel >> 8;
+        }
+    }
+    s->frame_offset = (s->frame_offset + len) % frame_bytes;
+}

+static void lcd_cam_capture(M5LcdCamState *s)
+{
+    uint32_t channel;
+    uint32_t len = s->chunk_len;
+    g_autofree uint8_t *data = g_malloc0(len);
+
+    if (!(s->int_ena & CAM_VSYNC_INT)) {
+        timer_mod(s->capture_timer,
+                  qemu_clock_get_ms(QEMU_CLOCK_VIRTUAL) + 10);
+        return;
+    }
+
+    lcd_cam_fill_frame(s, data, len);
+    if (s->gdma) {
+        bool found = esp_gdma_get_channel_periph(
+            s->gdma, GDMA_LCDCAM, ESP_GDMA_IN_IDX, &channel);
+        if (!found) {
+            if (getenv("M5SW_CAMERA_TRACE")) {
+                fprintf(stderr, "[lcd-cam] waiting for camera GDMA channel\n");
+            }
+            timer_mod(s->capture_timer,
+                      qemu_clock_get_ms(QEMU_CLOCK_VIRTUAL) + 10);
+            return;
+        }
+        if (s->chunks_left) {
+            esp_gdma_write_channel(s->gdma, channel, data, len);
+            esp_gdma_set_chan_interrupt(s->gdma, ESP_GDMA_IN_IDX, channel,
+                                        R_GDMA_INTERRUPT_IN_SUC_EOF_MASK);
+            s->chunks_left--;
+            timer_mod(s->capture_timer,
+                      qemu_clock_get_ms(QEMU_CLOCK_VIRTUAL) + 1);
+        } else {
+            s->int_raw |= CAM_VSYNC_INT;
+            lcd_cam_update_irq(s);
+            if (getenv("M5SW_CAMERA_TRACE")) {
+                fprintf(stderr, "[lcd-cam] frame complete bytes=%u\n",
+                        s->frame_bytes);
+            }
+        }
+    }
+}

+    switch (addr) {

+        break;
+    case R_CAM_CTRL1: {

+    }
+    case R_INT_ENA:

+        lcd_cam_update_irq(s);
+        break;
+    case R_INT_CLR:
+        s->int_raw &= ~v;
+        lcd_cam_update_irq(s);
+        break;
+    default:

+            s->regs[addr / 4] = v;
+        }
+        break;
+    }

+/*
+ * ESP32-S3 RMT APB and hardware-memory model.
+ *
+ * The UIFlow2 IR stack uses the new RMT driver in non-DMA mode: the encoder
+ * writes rmt_symbol_word_t entries into RMTMEM, starts the channel through
+ * CHnCONF0, and waits for the TX_END interrupt. This model implements that
+ * data path and records transmitted symbols for host-side assertions.
+ *
+ * SPDX-License-Identifier: GPL-2.0-or-later
+ */
+#include "qemu/osdep.h"
+#include "qemu/timer.h"
+#include "hw/irq.h"
+#include "hw/misc/esp32s3_reg.h"
+#include "hw/sysbus.h"
+#include "hw/xtensa/esp32s3_intc.h"
+#include "m5stopwatch.h"
+
+#define RMT_MEM_OFFSET      0x800
+#define RMT_MEM_SIZE        0x800
+#define RMT_TX_CHANNELS     4
+#define RMT_SYMBOLS_PER_CH  48
+
+#define RMT_CH0CONF0        0x20
+#define RMT_CH4CONF0        0x30
+#define RMT_CH4STATUS       0x60
+#define RMT_INT_RAW         0x70
+#define RMT_INT_ST          0x74
+#define RMT_INT_ENA         0x78
+#define RMT_INT_CLR         0x7c
+
+struct M5RmtState {
+    SysBusDevice parent_obj;
+    MemoryRegion mmio;
+    MemoryRegion mem;
+    qemu_irq irq;
+    uint32_t regs[RMT_MEM_OFFSET / 4];
+    uint32_t symbols[RMT_MEM_SIZE / 4];
+    uint32_t int_ena;
+    uint32_t int_raw;
+    QEMUTimer *tx_timer;
+    QEMUTimer *rx_timer;
+    FILE *tx_file;
+    uint32_t pending_mask;
+    uint32_t rx_pending_mask;
+    bool rx_injected[4];
+    bool trace;
+};
+
+static void rmt_update_irq(M5RmtState *s)
+{
+    qemu_set_irq(s->irq, !!(s->int_raw & s->int_ena));
+}
+
+static void rmt_capture_tx(M5RmtState *s, uint32_t channel)
+{
+    uint32_t conf = s->regs[(RMT_CH0CONF0 + channel * 4) / 4];
+    uint32_t blocks = MAX((conf >> 16) & 0xf, 1);
+    uint32_t count = 0;
+    uint32_t limit = MIN(blocks * RMT_SYMBOLS_PER_CH,
+                         RMT_SYMBOLS_PER_CH);
+    uint32_t base = channel * RMT_SYMBOLS_PER_CH;
+
+    for (uint32_t i = 0; i < limit; i++) {
+        uint32_t symbol = s->symbols[base + i];
+        if ((symbol & 0x7fff) == 0
+            && ((symbol >> 16) & 0x7fff) == 0) {
+            break;
+        }
+        count++;
+    }
+    if (!s->tx_file) {
+        const char *path = getenv("M5SW_RMT_TX_FILE");
+        if (path) {
+            s->tx_file = fopen(path, "wb");
+        }
+    }
+    if (s->tx_file && count) {
+        fwrite(&s->symbols[base], sizeof(uint32_t), count, s->tx_file);
+        fflush(s->tx_file);
+    }
+    if (s->trace) {
+        fprintf(stderr, "[rmt] tx channel=%u symbols=%u blocks=%u\n",
+                channel, count, blocks);
+    }
+
+    s->int_raw |= 1u << channel;
+    rmt_update_irq(s);
+}
+
+static void rmt_tx_timer(void *opaque)
+{
+    M5RmtState *s = opaque;
+    uint32_t pending = s->pending_mask;
+    s->pending_mask = 0;
+
+    for (uint32_t channel = 0; channel < RMT_TX_CHANNELS; channel++) {
+        if (pending & (1u << channel)) {
+            rmt_capture_tx(s, channel);
+        }
+    }
+}
+
+static void rmt_rx_timer(void *opaque)
+{
+    M5RmtState *s = opaque;
+    uint32_t pending = s->rx_pending_mask;
+    s->rx_pending_mask = 0;
+
+    for (uint32_t channel = 0; channel < 4; channel++) {
+        if (pending & (1u << channel)) {
+            s->int_raw |= 1u << (16 + channel);
+        }
+    }
+    rmt_update_irq(s);
+}
+
+static void rmt_inject_rx(M5RmtState *s, uint32_t channel)
+{
+    const char *path = getenv("M5SW_RMT_RX_FILE");
+    g_autofree uint8_t *data = NULL;
+    gsize size = 0;
+    uint32_t count;
+
+    if (!path || s->rx_injected[channel]) {
+        return;
+    }
+    if (!g_file_get_contents(path, (gchar **)&data, &size, NULL)) {
+        return;
+    }
+    count = MIN(size / sizeof(uint32_t), RMT_SYMBOLS_PER_CH);
+    memcpy(&s->symbols[(4 + channel) * RMT_SYMBOLS_PER_CH],
+           data, count * sizeof(uint32_t));
+    /* mem_waddr_ex is an absolute RMTMEM word address; each channel
+     * occupies 48 words on ESP32-S3 and starts at (channel + 4) * 48. */
+    s->regs[(RMT_CH4STATUS + channel * 4) / 4] =
+        (channel + 4) * 48 + count;
+    s->rx_injected[channel] = true;
+    s->rx_pending_mask |= 1u << channel;
+    timer_mod(s->rx_timer,
+              qemu_clock_get_ms(QEMU_CLOCK_REALTIME) + 20);
+    if (s->trace) {
+        fprintf(stderr, "[rmt] rx inject channel=%u symbols=%u\n",
+                channel, count);
+    }
+}
+
+static uint64_t rmt_read(void *opaque, hwaddr addr, unsigned size)
+{
+    M5RmtState *s = opaque;
+
+    if (addr == RMT_INT_RAW) {
+        return s->int_raw;
+    }
+    if (addr == RMT_INT_ST) {
+        return s->int_raw & s->int_ena;
+    }
+    if (addr == RMT_INT_ENA) {
+        return s->int_ena;
+    }
+    return addr < RMT_MEM_OFFSET ? s->regs[addr / 4] : 0;
+}
+
+static void rmt_write(void *opaque, hwaddr addr, uint64_t value,
+                      unsigned size)
+{
+    M5RmtState *s = opaque;
+    uint32_t v = value;
+
+    if (s->trace) {
+        fprintf(stderr, "[rmt] wr %03x <- %08x\n", (unsigned)addr, v);
+    }
+    if (addr == RMT_INT_ENA) {
+        s->int_ena = v;
+        rmt_update_irq(s);
+        return;
+    }
+    if (addr == RMT_INT_CLR) {
+        s->int_raw &= ~v;
+        rmt_update_irq(s);
+        return;
+    }
+    if (addr >= RMT_CH0CONF0 && addr < RMT_CH0CONF0 + RMT_TX_CHANNELS * 4) {
+        uint32_t channel = (addr - RMT_CH0CONF0) / 4;
+        s->regs[addr / 4] = v & ~1u;
+        if (v & 1u) {
+            s->pending_mask |= 1u << channel;
+            timer_mod(s->tx_timer,
+                      qemu_clock_get_ms(QEMU_CLOCK_REALTIME) + 2);
+        }
+        return;
+    }
+    if (addr >= RMT_CH4CONF0 && addr < RMT_CH4CONF0 + 4 * 8) {
+        uint32_t offset = addr - RMT_CH4CONF0;
+        uint32_t channel = offset / 8;
+        if ((offset % 8) == 4) {
+            bool was_enabled = s->regs[addr / 4] & 1u;
+            s->regs[addr / 4] = v;
+            if (!was_enabled && (v & 1u)) {
+                rmt_inject_rx(s, channel);
+            }
+            return;
+        }
+    }
+    if (addr < RMT_MEM_OFFSET) {
+        s->regs[addr / 4] = v;
+    }
+}
+
+static const MemoryRegionOps rmt_ops = {
+    .read = rmt_read,
+    .write = rmt_write,
+    .endianness = DEVICE_LITTLE_ENDIAN,
+    .valid.min_access_size = 4,
+    .valid.max_access_size = 4,
+    .impl.min_access_size = 4,
+    .impl.max_access_size = 4,
+};
+
+static void rmt_reset(DeviceState *dev)
+{
+    M5RmtState *s = M5_RMT(dev);
+    memset(s->regs, 0, sizeof(s->regs));
+    memset(s->symbols, 0, sizeof(s->symbols));
+    s->int_ena = 0;
+    s->int_raw = 0;
+    s->pending_mask = 0;
+    s->rx_pending_mask = 0;
+    memset(s->rx_injected, 0, sizeof(s->rx_injected));
+    timer_del(s->tx_timer);
+    timer_del(s->rx_timer);
+}
+
+static void rmt_init(Object *obj)
+{
+    M5RmtState *s = M5_RMT(obj);
+    SysBusDevice *sbd = SYS_BUS_DEVICE(obj);
+
+    memory_region_init_io(&s->mmio, obj, &rmt_ops, s, TYPE_M5_RMT,
+                          RMT_MEM_OFFSET);
+    memory_region_init_ram_ptr(&s->mem, obj, TYPE_M5_RMT ".mem",
