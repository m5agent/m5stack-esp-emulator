+static bool esp32c6_wifi_queue_rx(Esp32C6WifiState *s,
+                                  const uint8_t *frame, size_t length)
+{
+    uint8_t next = (s->rx_queue_tail + 1) % ARRAY_SIZE(s->rx_queue);
+    Esp32C6WifiPendingFrame *pending;
+
+    if (next == s->rx_queue_head || length > 1600) {
+        return false;
+    }
+    pending = &s->rx_queue[s->rx_queue_tail];
+    pending->length = length;
+    memcpy(pending->data, frame, length);
+    s->rx_queue_tail = next;
+    if (s->beacon_timer) {
+        timer_mod(s->beacon_timer,
+                  qemu_clock_get_ns(QEMU_CLOCK_VIRTUAL) + 5 * SCALE_MS);
+    }
+    return true;
+}

+static size_t esp32c6_wifi_build_probe_response(uint8_t *frame,
+                                                const uint8_t *station)
+{
+    static const uint8_t ap_mac[] = C6_WIFI_AP_MAC;
+    const char ssid[] = "myssid";
+    uint32_t now = qemu_clock_get_us(QEMU_CLOCK_VIRTUAL);
+    size_t length = 36;
+
+    memset(frame, 0, 128);
+    frame[0] = 0x50;
+    memcpy(frame + 4, station, 6);
+    memcpy(frame + 10, ap_mac, 6);
+    memcpy(frame + 16, ap_mac, 6);
+    frame[24] = now;
+    frame[25] = now >> 8;
+    frame[26] = now >> 16;
+    frame[27] = now >> 24;
+    frame[32] = 100;
+    frame[34] = 1;
+    frame[36] = 0;
+    frame[37] = sizeof(ssid) - 1;
+    memcpy(frame + 38, ssid, sizeof(ssid) - 1);
+    length += 2 + sizeof(ssid) - 1;
+    frame[length++] = 1;
+    frame[length++] = 4;
+    frame[length++] = 0x82;
+    frame[length++] = 0x84;
+    frame[length++] = 0x8b;
+    frame[length++] = 0x96;
+    frame[length++] = 3;
+    frame[length++] = 1;
+    frame[length++] = 1;
+    return length;
+}

+{
+    Esp32C6WifiState *s = ESP32C6_WIFI(opaque);
+    uint32_t value;

+static uint64_t esp32c6_wifi_phy_read(void *opaque, hwaddr addr,
+                                     unsigned size)
+{
+    Esp32C6WifiState *s = ESP32C6_WIFI(opaque);
+    uint32_t value = s->phy_regs[addr / sizeof(uint32_t)];
+
+    /*
+     * The C6 PHY treats set/clear transitions in FE_DATA as commands.  The
+     * TX-DC calibration polls bit 22 after starting a calibration with bit 0.
+     * QEMU completes calibration immediately, so expose the done bit as soon
+     * as the start bit is observed.
+     */
+    if (addr == 0x418) {
+        if (value & BIT(0)) {
+            value |= BIT(22);
+        } else {
+            value &= ~BIT(22);
+        }
+    } else if (addr == 0x814) {
+        /*
+         * PWDET reports the SAR state machine values 7 (idle) in bits 16:14.
+         * Calibration and conversion complete synchronously in this model.
+         */
+        value |= 0x1c000;
+    } else if (addr == 0x4a0) {
+        /* IQ estimation completes synchronously. */
+        value |= BIT(16);
+    } else if (addr == 0x0cc) {
+        /* RF frequency updates are instantaneous in this model. */
+        value |= BIT(8);
+    }
+
+    if (wifi_trace_enabled()) {
+        fprintf(stderr, "[c6-wifi-phy] rd %05x -> %08x\n",
+                (unsigned)addr, value);
+    }
+    return value;
+}

+static uint64_t esp32c6_wifi_mac_read(void *opaque, hwaddr addr,
+                                     unsigned size)
+{
+    Esp32C6WifiState *s = ESP32C6_WIFI(opaque);
+    uint32_t value = s->mac_regs[addr / sizeof(uint32_t)];
+
+    if (addr == 0x5c) {
+        value = (value & 0xffff0000) |
+                (uint32_t)s->macaddr[0] |
+                ((uint32_t)s->macaddr[1] << 8) |
+                ((uint32_t)s->macaddr[2] << 16) |
+                ((uint32_t)s->macaddr[3] << 24);
+    } else if (addr == 0x60) {
+        value = (value & 0xffff0000) |
+                (uint32_t)s->macaddr[4] |
+                ((uint32_t)s->macaddr[5] << 8);
+    } else if (addr == 0xc48) {
+        value = s->raw_interrupt;
+    } else if (addr == 0x80 && (value & BIT(0))) {
+        /*
+         * RX_DSCR_RELOAD is a self-clearing hardware status bit.  Reads
+         * observe the request once, then the MAC clears it.
+         */
+        s->mac_regs[addr / sizeof(uint32_t)] &= ~BIT(0);
+        value &= ~BIT(0);
+    }
+
+    /*
+     * MAC reset is acknowledged synchronously.  The HAL writes bit 1 to
+     * MAC_RESET and waits for the completion bit 0.
+     */
+    if (addr == 0xddc && (value & BIT(1))) {
+        value |= BIT(0);
+    }
+
+    if (wifi_trace_enabled()) {
+        fprintf(stderr, "[c6-wifi-mac] rd %05x -> %08x\n",
+                (unsigned)addr, value);
+    }
+    return value;
+}

+static void esp32c6_wifi_mac_write(void *opaque, hwaddr addr,
+                                   uint64_t value64, unsigned size)
+{
+    Esp32C6WifiState *s = ESP32C6_WIFI(opaque);
+    uint32_t value = value64;
+
+    if (wifi_trace_enabled()) {
+        fprintf(stderr, "[c6-wifi-mac] wr %05x <- %08x\n",
+                (unsigned)addr, value);
+        if (addr == 0x84 && value) {
+            uint32_t words[3] = { 0 };
+            uint32_t first[4] = { 0 };
+
+            address_space_read(&address_space_memory, value,
+                               MEMTXATTRS_UNSPECIFIED, words, sizeof(words));
+            fprintf(stderr,
+                    "[c6-wifi-mac] rx head=%08x desc=%08x,%08x,%08x\n",
+                    value, words[0], words[1], words[2]);
+            if (words[1]) {
+                address_space_read(&address_space_memory, words[1],
+                                   MEMTXATTRS_UNSPECIFIED, first,
+                                   sizeof(first));
+                fprintf(stderr,
+                        "[c6-wifi-mac] rx buffer=%08x data=%08x,%08x,%08x,%08x\n",
+                        words[1], first[0], first[1], first[2], first[3]);
+            }
+        }
+    }
+    if (addr == 0x84) {
+        s->rx_ring_head = value;
+        s->rx_dma_ptr = value;
+        timer_mod(s->beacon_timer,
+                  qemu_clock_get_ns(QEMU_CLOCK_VIRTUAL) + 50 * SCALE_MS);
+    } else if (addr == 0xd6c && (value & 0xc0000000)) {
+        uint32_t descriptor = 0x40800000u | (value & 0x000fffffu);
+        uint32_t words[8] = { 0 };
+        uint8_t tx_data[2048] = { 0 };
+
+        address_space_read(&address_space_memory, descriptor,
+                           MEMTXATTRS_UNSPECIFIED, words, sizeof(words));
+        if (wifi_trace_enabled()) {
+            fprintf(stderr,
+                    "[c6-wifi-mac] TX trigger=%08x descriptor=%08x "
+                    "words=%08x,%08x,%08x,%08x,%08x,%08x,%08x,%08x\n",
+                    (unsigned)value, descriptor,
+                    words[0], words[1], words[2], words[3],
+                    words[4], words[5], words[6], words[7]);
+        }
+        if (words[1]) {
+            uint8_t response[256] = { 0 };
+
+            address_space_read(&address_space_memory, words[1],
+                               MEMTXATTRS_UNSPECIFIED, tx_data,
+                               sizeof(tx_data));
+            if (wifi_trace_enabled()) {
+                fprintf(stderr, "[c6-wifi-mac] TX data");
+                for (unsigned i = 0; i < 128; i++) {
+                    fprintf(stderr, " %02x", tx_data[i]);
+                }
+                fprintf(stderr, "\n");
+            }
+
+            {
+                size_t frame_length = ldl_le_p(tx_data);
+                uint8_t *mac = tx_data + 8;
+                uint16_t fc = mac[0] | (mac[1] << 8);
+                unsigned type = (fc >> 2) & 3;
+                unsigned subtype = (fc >> 4) & 0xf;
+
+                if (frame_length > sizeof(tx_data) - 8) {
+                    frame_length = sizeof(tx_data) - 8;
+                }
+                if (!memcmp(mac + 10, "\0\0\0\0\0\0", 6)) {
+                    memcpy(mac + 10, s->macaddr, 6);
+                }
+                if (wifi_trace_enabled()) {
+                    fprintf(stderr,
+                            "[c6-wifi-mac] TX frame type=%u subtype=%u "
+                            "len=%zu\n", type, subtype, frame_length);
+                }
+                if (type == 0 && subtype == 4) {
+                    size_t response_length =
+                        esp32c6_wifi_build_probe_response(response,
+                                                          mac + 10);
+                    esp32c6_wifi_queue_rx(s, response, response_length);
+                } else if (type == 0 && subtype == 11) {
+                    static const uint8_t ap_mac[] = C6_WIFI_AP_MAC;
+
+                    response[0] = 0xb0;
+                    memcpy(response + 4, mac + 10, 6);
+                    memcpy(response + 10, ap_mac, 6);
+                    memcpy(response + 16, ap_mac, 6);
+                    response[26] = 2;
+                    esp32c6_wifi_queue_rx(s, response, 30);
+                } else if (type == 0 && subtype == 0) {
+                    static const uint8_t ap_mac[] = C6_WIFI_AP_MAC;
+
+                    response[0] = 0x10;
+                    memcpy(response + 4, mac + 10, 6);
+                    memcpy(response + 10, ap_mac, 6);
+                    memcpy(response + 16, ap_mac, 6);
+                    response[24] = 1;
+                    response[28] = 1;
+                    response[29] = 0xc0;
+                    response[30] = 1;
+                    response[31] = 4;
+                    response[32] = 0x82;
+                    response[33] = 0x84;
+                    response[34] = 0x8b;
+                    response[35] = 0x96;
+                    s->associated = true;
+                    esp32c6_wifi_queue_rx(s, response, 36);
+                } else if (type == 0 &&
+                           (subtype == 10 || subtype == 12)) {
+                    s->associated = false;
+                } else if (type == 2 && s->associated) {
+                    uint8_t ethernet[1518] = { 0 };
+                    size_t payload_length =
+                        frame_length > 36 ? frame_length - 36 : 0;
+
+                    if (payload_length > sizeof(ethernet) - 14) {
+                        payload_length = sizeof(ethernet) - 14;
+                    }
+                    memcpy(ethernet, mac + 16, 6);
+                    memcpy(ethernet + 6, s->macaddr, 6);
+                    memcpy(ethernet + 12, mac + 30,
+                           payload_length + 2);
+                    ssize_t sent =
+                        qemu_send_packet(qemu_get_queue(s->nic), ethernet,
+                                         payload_length + 14);
+                    if (wifi_trace_enabled()) {
+                        fprintf(stderr,
+                                "[c6-wifi-mac] Ethernet TX len=%zu ret=%zd\n",
+                                payload_length + 14, sent);
+                        if (payload_length + 14 >= 42) {
+                            fprintf(stderr, "[c6-wifi-mac] Ethernet data:");
+                            for (size_t i = 0; i < 42; i++) {
+                                fprintf(stderr, " %02x", ethernet[i]);
+                            }
+                            fprintf(stderr, "\n");
+                        }
+                    }
+                }
+            }
+        }
+        /*
+         * DMA completion is reported through the MAC TX-state registers, not
+         * by the interrupt bit alone.  Queue 2 carries management frames and
+         * slot 0 is the currently used TX descriptor.  Returning zero here
+         * leaves the firmware lmac queue permanently busy after the first
+         * probe request, so it queues Authentication but never programs a
+         * second TX descriptor.
+         */
+        s->mac_regs[0xc34 / sizeof(uint32_t)] |= BIT(0);
+        s->mac_regs[0xcb8 / sizeof(uint32_t)] |= BIT(0);
+        s->raw_interrupt |= C6_WIFI_IRQ_TX_DONE;
+        qemu_irq_raise(s->irq);
+    } else if (addr == 0xc4c) {
+        s->raw_interrupt &= ~value;
+        if (!s->raw_interrupt) {
+            qemu_irq_lower(s->irq);
+        }
+        return;
+    } else if (addr == 0xcb4) {
+        s->mac_regs[0xcb8 / sizeof(uint32_t)] &= ~(value & 0x7ff);
+    }
+    s->mac_regs[addr / sizeof(uint32_t)] = value;
+    if (addr == 0x5c && value) {
+        memcpy(s->macaddr, &value, 4);
+    } else if (addr == 0x60 && (value & 0xffff)) {
+        s->macaddr[4] = value;
+        s->macaddr[5] = value >> 8;
+    }
+}

+static void esp32c6_wifi_beacon_timer(void *opaque)
+{
+    Esp32C6WifiState *s = ESP32C6_WIFI(opaque);
+    uint8_t packet[C6_WIFI_RX_CTRL_LEN + 1600] = { 0 };
+    uint8_t *frame = packet + C6_WIFI_RX_CTRL_LEN;
+    uint32_t descriptor = s->rx_dma_ptr;
+    uint32_t words[3] = { 0 };
+    uint32_t next = 0;
+    size_t frame_len = 0;
+    size_t total;
+    uint32_t word0;
+    bool queued = false;
+    bool group = false;
+
+    if (!descriptor || s->raw_interrupt) {
+        goto reschedule;
+    }
+    if (address_space_read(&address_space_memory, descriptor,
+                           MEMTXATTRS_UNSPECIFIED, words,
+                           sizeof(words)) != MEMTX_OK) {
+        goto reschedule;
+    }
+    if (!words[1] || !(words[0] & BIT(31))) {
+        goto reschedule;
+    }
+    next = words[2] ? words[2] : s->rx_ring_head;
+
+    if (s->rx_queue_head != s->rx_queue_tail) {
+        Esp32C6WifiPendingFrame *pending =
+            &s->rx_queue[s->rx_queue_head];
+
+        frame_len = pending->length;
+        memcpy(frame, pending->data, frame_len);
+        s->rx_queue_head =
+            (s->rx_queue_head + 1) % ARRAY_SIZE(s->rx_queue);
+        queued = true;
+    } else {
+        static const uint8_t ap_mac[] = C6_WIFI_AP_MAC;
+        const char ssid[] = "myssid";
+        uint32_t now = qemu_clock_get_us(QEMU_CLOCK_VIRTUAL);
+
+        frame[0] = (s->beacon_count & 1) ? 0x50 : 0x80;
+        memset(frame + 4, 0xff, 6);
+        memcpy(frame + 10, ap_mac, 6);
+        memcpy(frame + 16, ap_mac, 6);
+        frame[24] = now;
+        frame[25] = now >> 8;
+        frame[26] = now >> 16;
+        frame[27] = now >> 24;
+        frame[32] = 100;
+        frame[34] = 1;
+        frame_len = 36;
+        frame[37] = sizeof(ssid) - 1;
+        memcpy(frame + 38, ssid, sizeof(ssid) - 1);
+        frame_len += 2 + sizeof(ssid) - 1;
+        frame[frame_len++] = 1;
+        frame[frame_len++] = 4;
+        frame[frame_len++] = 0x82;
+        frame[frame_len++] = 0x84;
+        frame[frame_len++] = 0x8b;
+        frame[frame_len++] = 0x96;
+        frame[frame_len++] = 3;
+        frame[frame_len++] = 1;
+        frame[frame_len++] = 1;
+    }
+    /* FCS is appended to every injected MPDU. */
+    memset(frame + frame_len, 0, 4);
+    frame_len += 4;
+    group = !memcmp(frame + 4, "\xff\0\0\0\0\0", 6);
+
+    packet[0] = 0xdd;
+    packet[1] = 11;
+    packet[3] = group ? 0xf0 : 0x10;
+    packet[11] = group ? 0x80 : 0;
+    packet[12] = qemu_clock_get_us(QEMU_CLOCK_VIRTUAL);
+    packet[13] = qemu_clock_get_us(QEMU_CLOCK_VIRTUAL) >> 8;
+    packet[14] = qemu_clock_get_us(QEMU_CLOCK_VIRTUAL) >> 16;
+    packet[15] = qemu_clock_get_us(QEMU_CLOCK_VIRTUAL) >> 24;
+    packet[20] = 0x9f;
+    packet[21] = 1;
+    packet[39] = 0;
+    packet[84] = frame_len;
+    packet[85] = frame_len >> 8;
+    packet[86] = frame_len - 4;
+    packet[87] = (frame_len - 4) >> 8;
+
+    total = C6_WIFI_RX_CTRL_LEN + frame_len;
+    if ((words[0] & 0x3fff) < total) {
+        goto reschedule;
+    }
+    if (address_space_write(&address_space_memory, words[1],
+                            MEMTXATTRS_UNSPECIFIED, packet,
+                            total) != MEMTX_OK) {
+        goto reschedule;
+    }
+    if (wifi_trace_enabled()) {
+        fprintf(stderr, "[c6-wifi-mac] RX data %s len=%zu:",
+                queued ? "frame" : "beacon", frame_len);
+        for (size_t i = 0; i < MIN(frame_len, 64); i++) {
+            fprintf(stderr, " %02x", frame[i]);
+        }
+        fprintf(stderr, "\n");
+    }
+
+    word0 = (words[0] & 0x3fff) | ((uint32_t)total << 14) | BIT(30);
+    address_space_write(&address_space_memory, descriptor,
+                        MEMTXATTRS_UNSPECIFIED, &word0, sizeof(word0));
+    s->mac_regs[0x8c / sizeof(uint32_t)] = descriptor & 0x000fffff;
+    s->mac_regs[0xc70 / sizeof(uint32_t)] = descriptor & 0xfff00000;
+    s->mac_regs[0x88 / sizeof(uint32_t)] = next;
+    s->rx_dma_ptr = next;
+    s->raw_interrupt |= C6_WIFI_IRQ_RX_DONE;
+    s->beacon_count++;
+    qemu_irq_raise(s->irq);
+    if (wifi_trace_enabled()) {
+        fprintf(stderr,
+                "[c6-wifi-mac] RX %s descriptor=%08x next=%08x "
+                "len=%zu\n", queued ? "frame" : "beacon",
+                descriptor, next, total);
+    }
+
+reschedule:
+    if (s->rx_dma_ptr) {
+        timer_mod(s->beacon_timer,
+                  qemu_clock_get_ns(QEMU_CLOCK_VIRTUAL) +
+                  (s->rx_queue_head != s->rx_queue_tail ?
+                   5 * SCALE_MS : 50 * SCALE_MS));
+    }
+}

+static bool esp32c6_wifi_can_receive(NetClientState *ncs)
+{
+    Esp32C6WifiState *s = qemu_get_nic_opaque(ncs);
+    uint8_t next = (s->rx_queue_tail + 1) % ARRAY_SIZE(s->rx_queue);
+
+    return s->associated && next != s->rx_queue_head;
+}

+static ssize_t esp32c6_wifi_receive(NetClientState *ncs,
+                                    const uint8_t *buf, size_t size)
+{
+    Esp32C6WifiState *s = qemu_get_nic_opaque(ncs);
+    static const uint8_t ap_mac[] = C6_WIFI_AP_MAC;
+    uint8_t frame[1600] = { 0 };
+    size_t frame_length;
+    uint16_t sequence;
+
+    if (size < 14 || size > 1518) {
+        return -1;
+    }
+    frame_length = 24 + 6 + size - 12;
+    if (frame_length > sizeof(frame)) {
+        return -1;
+    }
+
+    frame[0] = 0x08;
+    frame[1] = 0x02;
+    frame[2] = 44;
+    memcpy(frame + 4, buf, 6);
+    memcpy(frame + 10, ap_mac, 6);
+    memcpy(frame + 16, buf + 6, 6);
+    sequence = s->sequence++ & 0xfff;
+    frame[22] = sequence << 4;
+    frame[23] = sequence >> 4;
+    frame[24] = 0xaa;
+    frame[25] = 0xaa;
+    frame[26] = 0x03;
+    memcpy(frame + 30, buf + 12, size - 12);
+    if (!esp32c6_wifi_queue_rx(s, frame, frame_length)) {
+        return -1;
+    }
+    if (wifi_trace_enabled()) {
+        fprintf(stderr,
+                "[c6-wifi-mac] Ethernet RX len=%zu -> 802.11 len=%zu\n",
+                size, frame_length);
+    }
+    return size;
+}
