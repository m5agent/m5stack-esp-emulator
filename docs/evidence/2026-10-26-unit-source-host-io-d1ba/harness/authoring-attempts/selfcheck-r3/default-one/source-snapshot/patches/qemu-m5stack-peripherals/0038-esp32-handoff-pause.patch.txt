+static void esp32_handoff_pause_start(void)
+{
+    const char *value = getenv("M5SW_ESP32_HANDOFF_PAUSE_NS");
+    uint64_t pause_ns;
+
+    if (!esp32_handoff_trace_enabled() || !value) {

+    }
+    timer_mod_ns(handoff_pause_timer, pause_ns);
