+static bool xtensa_handoff_pc_trace_match(uint32_t pc)
+{
+    static uint32_t trace_pcs[32];
+    static unsigned int trace_pc_count;
+    static bool initialized;
+
+    if (!initialized) {
+        const char *value = getenv("M5SW_XTENSA_PC_TRACE");
+
+        if (value) {
+            gchar **parts = g_strsplit(value, ",", -1);
+
+            for (unsigned int i = 0;
+                 parts[i] && trace_pc_count < ARRAY_SIZE(trace_pcs); ++i) {
+                char *end = NULL;
+                char *token = g_strstrip(parts[i]);
+                unsigned long long address;
+
+                if (!*token) {
+                    continue;
+                }
+                address = strtoull(token, &end, 0);
+                if (end && !*end) {
+                    trace_pcs[trace_pc_count++] = address;
+                }
+            }
+            g_strfreev(parts);
+        }
+        initialized = true;
+    }
+
+    for (unsigned int i = 0; i < trace_pc_count; ++i) {
+        if (trace_pcs[i] == pc) {
+            return true;
+        }
+    }
+    return false;
+}
