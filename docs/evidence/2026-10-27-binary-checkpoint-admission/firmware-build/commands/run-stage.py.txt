#!/usr/bin/env python3
"""Run one owned, bounded build stage and retain its exact process result."""
from pathlib import Path
import argparse,datetime,json,os,signal,subprocess,time
p=argparse.ArgumentParser();p.add_argument('--name',required=True);p.add_argument('--timeout',type=int,required=True);p.add_argument('--cwd');p.add_argument('command',nargs=argparse.REMAINDER);a=p.parse_args()
base=Path(__file__).resolve().parent
logs=base/'logs';logs.mkdir(exist_ok=True)
record=logs/(a.name+'.json');log=logs/(a.name+'.log')
if record.exists():raise SystemExit('stage already exists; inspect its terminal state before choosing a new stage name')
env=dict(os.environ)
env.update(IDF_PATH=str(base/'idf'),IDF_CCACHE_ENABLE='0',IDF_COMPONENT_CACHE_PATH=str(base/'cache/components'),IDF_TOOLS_PATH=str(base/'idf-tools'),IDF_PYTHON_ENV_PATH=str(base/'idf-python-env'),PIP_CACHE_DIR=str(base/'cache/pip'),XDG_CACHE_HOME=str(base/'cache'),PYTHONNOUSERSITE='1')
env['PATH']=str(base/'host-tools/bin')+os.pathsep+str(base/'host-env/bin')+os.pathsep+env.get('PATH','')
start=time.time(); info={'name':a.name,'argv':a.command,'cwd':a.cwd or str(base),'started_at':datetime.datetime.now(datetime.timezone.utc).isoformat(),'timeout_seconds':a.timeout,'state':'starting'}
def save():
 temp=record.with_suffix('.tmp');temp.write_text(json.dumps(info,indent=2)+'\n');temp.replace(record)
save()
with log.open('wb') as output:
 try:proc=subprocess.Popen(a.command,cwd=a.cwd or base,env=env,stdout=output,stderr=subprocess.STDOUT,start_new_session=True)
 except OSError as error:
  info.update(state='finished',exit_code=127,spawn_failed=True,error=str(error),elapsed_seconds=round(time.time()-start,3));save();raise
 info.update(pid=proc.pid,pgid=proc.pid,state='running');save();print(json.dumps({'stage':a.name,'pid':proc.pid,'log':str(log)}),flush=True)
 try:code=proc.wait(timeout=a.timeout)
 except subprocess.TimeoutExpired:
  info['timed_out']=True;os.killpg(proc.pid,signal.SIGTERM)
  try:code=proc.wait(timeout=10)
  except subprocess.TimeoutExpired:os.killpg(proc.pid,signal.SIGKILL);code=proc.wait(timeout=10)
 info.update(state='finished',exit_code=code,elapsed_seconds=round(time.time()-start,3),finished_at=datetime.datetime.now(datetime.timezone.utc).isoformat());save()
print(json.dumps(info),flush=True)
raise SystemExit(code if code>=0 else 128-code)
