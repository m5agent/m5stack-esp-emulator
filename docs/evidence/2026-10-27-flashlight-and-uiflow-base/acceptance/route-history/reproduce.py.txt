#!/usr/bin/env python3
"""Compile the actual 0045 model with existing GPIO stubs; no guest or API."""
import hashlib
import importlib.util
import json
from pathlib import Path
import shutil
import subprocess
import sys
import tempfile

HERE = Path(__file__).resolve().parent
ROOT = HERE.parents[1]
PATCH = ROOT / "patches/qemu-m5stack-peripherals/0045-unit-flashlight-aw3641.patch"
TEST = ROOT / "platform/tests/test_unit_flashlight_patch.py"
patch_bytes, test_bytes = PATCH.read_bytes(), TEST.read_bytes()
spec = importlib.util.spec_from_file_location("route_history_existing_stubs", TEST)
module = importlib.util.module_from_spec(spec)
spec.loader.exec_module(module)
added = "\n".join(line[1:] for line in patch_bytes.decode().splitlines()
                  if line.startswith("+") and not line.startswith("+++"))
start = added.index("/* BEGIN AW3641 NOMINAL DIGITAL MODEL")
end = added.index("/* END AW3641 NOMINAL DIGITAL MODEL */") + len("/* END AW3641 NOMINAL DIGITAL MODEL */")
fragment = added[start:end]
hook = added[added.index("void m5_unit_gpio_initial_complete(M5Unit *unit)"):]
hook = hook[:hook.index("\n}") + len("\n}")]
helpers = module.TESTS[:module.TESTS.index("static void table(void)")].replace("/* FREEZE_HOOK */", hook)
experiment = r'''
int main(void) {
 M5Unit u; AW3641State *s=create(&u,false,true,false);
 reset_low(s); pad(1); stable(s);
 assert(get_bool("output-known") && get_bool("active"));
 uint64_t edges=s->edge_count;
 routes[2]=73;
 bool invalid_known=get_bool("output-known"), invalid_active=get_bool("active");
 routes[2]=256;
 printf("{\"unsupported_known\":%s,\"unsupported_active\":%s,\"restored_known_without_reset\":%s,\"restored_active_without_reset\":%s,\"edge_count_unchanged\":%s}\n",
 invalid_known?"true":"false", invalid_active?"true":"false",
 get_bool("output-known")?"true":"false", get_bool("active")?"true":"false",
 s->edge_count==edges?"true":"false");
 destroy(s); return 0;
}
'''
source = (module.STUBS + fragment + helpers + experiment).encode()
(HERE / "harness.c").write_bytes(source)
cc = shutil.which("cc")
assert cc, "C compiler unavailable"
with tempfile.TemporaryDirectory(prefix="m5-route-history-audit-") as temporary:
    binary = Path(temporary) / "route-history"
    compile_command = [cc, "-std=c11", "-Wall", "-Wextra", "-Werror",
                       "-Wno-unused-parameter", "-Wno-unused-function", str(HERE / "harness.c"), "-o", str(binary)]
    compiled = subprocess.run(compile_command, capture_output=True, timeout=30)
    assert compiled.returncode == 0, compiled.stderr.decode()
    executed = subprocess.run([str(binary)], capture_output=True, timeout=5)
    assert executed.returncode == 0, executed.stderr.decode()
    assert json.loads(executed.stdout) == {
        "unsupported_known": False, "unsupported_active": False,
        "restored_known_without_reset": True, "restored_active_without_reset": True,
        "edge_count_unchanged": True,
    }
    (HERE / "stdout.txt").write_bytes(executed.stdout)
    (HERE / "stderr.txt").write_bytes(compiled.stderr + executed.stderr)
    commands = {"reproduce_argv": [sys.executable, "-B", str(Path(__file__).resolve())],
                "cwd": str(ROOT), "compile_argv": compile_command, "execute_argv": [str(binary)],
                "compile_exit_code": compiled.returncode, "execute_exit_code": executed.returncode,
                "binary_location": "TemporaryDirectory only; removed after execution"}
assert PATCH.read_bytes() == patch_bytes and TEST.read_bytes() == test_bytes
(HERE / "commands.json").write_text(json.dumps(commands, indent=2) + "\n")
files = [PATCH, TEST, Path(__file__).resolve(), HERE / "harness.c", HERE / "stdout.txt", HERE / "stderr.txt"]
receipt = {"category": "static-controller-C-diagnostic", "result": "ROUTE_HISTORY_GAP_REPRODUCED",
           "api_contacted": False, "guest_execution": "NOT RUN", "hardware_execution": "NOT RUN",
           "scope": "Actual patch C with existing synthetic pad/route/virtual-clock stubs. No guest or physical-controller inference.",
           "observation": "After qualified steady Torch, unsupported route withdraws known. Restoring GPIO route 256 without an edge/reset regrants old known/active. Getter only ANDs current route/pad, without latching history.",
           "release_scope": "Unadmitted dynamic-route path; does not negate the recorded continuously observed GPIO subset.",
           "source_sha256": {str(p.relative_to(ROOT)): hashlib.sha256(p.read_bytes()).hexdigest() for p in files}}
(HERE / "receipt.json").write_text(json.dumps(receipt, indent=2) + "\n")
print(executed.stdout.decode().strip())
