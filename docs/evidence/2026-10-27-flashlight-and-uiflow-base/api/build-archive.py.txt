def check(raw):

 assert not re.search(rb'(?:sk-proj-|ghp_|gho_|xoxb-)[A-Za-z0-9_-]{10,}',raw),'credential pattern found'
 assert not re.search(rb'(?i)\b(?:[0-9a-f]{2}:){5}[0-9a-f]{2}\b',raw),'MAC found'
 raw.decode('utf8',errors='strict')

def write(relative,raw,*,source,kind='captured',extra=None):
 check(raw);path=DEST/relative;path.parent.mkdir(parents=True,exist_ok=True);path.write_bytes(raw)

def generated(relative,value):

def reconstructed(relative,raw,*,historical_sha,derivation,inputs):
