def sha(data):
    return hashlib.sha256(data).hexdigest()


def save(name, value):
    (OUT / name).write_text(json.dumps(value, ensure_ascii=False, indent=2) + "\n")


def run(name, args, *, environment=None):
    result = subprocess.run(args, cwd=ROOT, env=environment, stdout=subprocess.PIPE, stderr=subprocess.STDOUT)
    log = OUT / f"{name}.log"
    log.write_bytes(result.stdout)
    record = {"argv": args, "cwd": "repository-root", "environment_overrides":
              {"PYTHONPATH": environment["PYTHONPATH"]} if environment is not None else {},
              "exit_code": result.returncode, "stdout_stderr_file": log.name,
              "stdout_stderr_sha256": sha(result.stdout), "stdout_stderr": result.stdout.decode("utf-8")}
    save(f"{name}.json", record)
    if result.returncode:
        raise RuntimeError(f"{name} failed; inspect {log.name}")
    return record
