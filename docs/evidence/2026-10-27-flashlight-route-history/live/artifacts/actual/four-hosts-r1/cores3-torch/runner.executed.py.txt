#!/usr/bin/env python3
"""Prepare, or explicitly execute, actual UART/MMIO FlashLight lifecycle checks.

Default invocation is offline. This is a fresh preparation prototype, not a
record of a past execution. No QOM write or output-state injection exists here.
"""
from __future__ import annotations

import argparse
import base64
import io
import json
from pathlib import Path
import re
import sys
import time
import urllib.parse

ROOT = Path(__file__).resolve().parents[2]
sys.path[:0] = [str(ROOT / "tests/units"), str(ROOT / "platform/tools")]
import run_binary_checkpoint_admission as common

DRIVER = ROOT / "docs/evidence/2026-10-27-binary-checkpoint-admission/research/materials/uiflow-flash_light-2.5.3.py"
DRIVER_SHA = "10546386ee156034c268404e96f4d086b88c52254da1fcd2aa19130ce2f6629f"
PREFIX = "FLASHLIGHT_ROUTE_GUEST "
PORTS = {"cores3": "port-a", "atoms3r": "custom", "sticks3": "custom", "stopwatch": "port-a"}
ROUTES = {"ledc73": 73, "rmt81": 81, "out-inv": 256 | 512}
CASES = [(name, "roundtrip") for name in ROUTES] + [
    ("ledc73", "unsupported-low-hold"), ("ledc73", "unsupported-low-rise")]
PROPERTIES = ("output-known", "active", "waveform-qualified", "program-state", "unknown-reason",
              "pad-synchronized", "gpio-route-supported", "en-level", "flash-mode", "mode-frozen",
              "edge-count", "rising-edges", "decoded-programs", "reset-events", "timeout-events",
              "unqualified-programs", "settle-at-ns", "timeout-at-ns", "edge-trace-ns")


def guest_source(nonce, phase, pins, route_name="ledc73", pattern="roundtrip"):
    common.require(re.fullmatch("[0-9a-f]{32}", nonce), "invalid nonce")
    common.require(route_name in ROUTES and pattern in ("roundtrip", "unsupported-low-hold", "unsupported-low-rise"), "invalid case")
    common.require(phase in ("bootstrap", "atomic", "stable-unknown", "recover-reset", "recover-driver",
                             "web-away", "web-return", "pwm-companion", "pin-companion"), "invalid phase")
    state_key = "_m5_flash_route_" + nonce
    prelude = f'''import sys, os, json, utime, hashlib, binascii
import M5
from machine import Pin, mem32
import official_flash
with open(official_flash.__file__, 'rb') as _source_file:
 _driver_bytes = _source_file.read()
assert len(_driver_bytes) == 3639 and binascii.hexlify(hashlib.sha256(_driver_bytes).digest()).decode() == {DRIVER_SHA!r}
key = {state_key!r}
phase = {phase!r}
route_name = {route_name!r}
pattern = {pattern!r}
events = []
'''
    if phase == "bootstrap":
        action = f'''assert key not in sys.modules, 'state nonce reused'
M5.begin()
M5.Display.setBrightness(255)
M5.Display.fillScreen(0)
flash = official_flash.FlashLightUnit(({pins['white']}, {pins['yellow']}))
en = Pin({pins['yellow']})
reg = 0x60004000 + 0x554 + 4 * {pins['yellow']}
base = mem32[reg]
assert base & 0x3ff == 256, 'bootstrap GPIO route is unsupported'
st = {{'en': en, 'flash': flash, 'reg': reg, 'base': base, 'nonce': {nonce!r}}}
sys.modules[key] = st
en.value(1)
utime.sleep_us(2)
en.value(0)
utime.sleep_us(600)
'''
    else:
        action = f'''assert key in sys.modules, 'same-VM nonce disappeared'
st = sys.modules[key]
assert st['nonce'] == {nonce!r}
en, reg, base = st['en'], st['reg'], st['base']
'''
        if phase == "atomic":
            action += f'''assert mem32[reg] & 0x3ff == 256, 'prepare requires restored GPIO route'
en.value(1)
utime.sleep_us(2)
en.value(0)
utime.sleep_us(600)
arm_start = utime.ticks_us()
st['flash'].flash(st['flash'].BRIGHTNESS_100, st['flash'].TIME_220MS, False)
# Give the real guest scheduler/event loop a chance to deliver the model's
# 500us settle timer. sleep_us alone can busy-wait across its deadline.
# The measured total must still be below 100ms, before old Flash expiry.
utime.sleep_ms(20)
assert int(en.value()) == 1
before_raw = mem32[reg]
away_raw = (before_raw & ~0x3ff) | {ROUTES[route_name]}
switch_start = utime.ticks_us()
mem32[reg] = away_raw
away_readback = mem32[reg]
assert away_readback == away_raw, 'unsupported route readback differs'
'''
            if pattern != "roundtrip":
                action += "en.value(0)\nutime.sleep_us(800)\n"
            action += "mem32[reg] = before_raw\nreturn_readback = mem32[reg]\nassert return_readback == before_raw\n"
            if pattern == "unsupported-low-rise":
                # There must be no hold, serial, API, sleep or getter between
                # restoring the route and this immediate rise.
                action += "en.value(1)\n"
            action += '''switch_end = utime.ticks_us()
assert utime.ticks_diff(switch_end, arm_start) < 100000, 'missed old 220ms Flash deadline window'
events.append({'before_raw': before_raw, 'away_raw': away_readback, 'returned_raw': return_readback,
               'arm_to_return_us': utime.ticks_diff(switch_end, arm_start),
               'route_window_us': utime.ticks_diff(switch_end, switch_start)})
utime.sleep_us(600)
'''
        elif phase == "stable-unknown":
            action += "utime.sleep_ms(240)\n"
        elif phase == "recover-reset":
            action += "assert mem32[reg] & 0x3ff == 256\nen.value(1)\nutime.sleep_us(2)\nen.value(0)\nutime.sleep_us(600)\n"
        elif phase == "recover-driver":
            # Sample Flash only after its nominal deadline, so independent
            # property GETs cannot race the expected active->expired timer.
            # The atomic case separately proves decoding before route return.
            action += "assert mem32[reg] & 0x3ff == 256\nst['flash'].flash(st['flash'].BRIGHTNESS_100, st['flash'].TIME_220MS, False)\nutime.sleep_ms(240)\n"
        elif phase == "web-away":
            action += f"assert mem32[reg] & 0x3ff == 256\nmem32[reg] = (mem32[reg] & ~0x3ff) | {ROUTES[route_name]}\nassert mem32[reg] & 0x3ff == {ROUTES[route_name]}\n"
        elif phase == "web-return":
            action += "mem32[reg] = base\nassert mem32[reg] & 0x3ff == 256\n"
        elif phase == "pwm-companion":
            action += '''from machine import PWM
pwm = PWM(en, freq=1000, duty_u16=32768)
events.append({'step': 'PWM-created', 'route_raw': mem32[reg]})
pwm.deinit()
events.append({'step': 'PWM-deinit', 'route_raw': mem32[reg]})
'''
        elif phase == "pin-companion":
            action += f"en = Pin({pins['yellow']}, Pin.OUT, value=0)\nst['en'] = en\nevents.append({{'step': 'Pin-init', 'route_raw': mem32[reg]}})\nmem32[reg] = base\n"
    return prelude + action + f'''route_raw = mem32[reg]
level = int(en.value())
M5.Display.fillRect(M5.Display.width()//2-4, M5.Display.height()//2-4, 8, 8, 0xFFFFFF if level else 0)
print({PREFIX!r} + json.dumps({{'nonce': {nonce!r}, 'phase': phase, 'route_name': route_name,
 'pattern': pattern, 'en': level, 'route_raw': route_raw, 'route_effective': -1 if route_raw & 512 else route_raw & 511,
 'events': events, 'driver_sha256': {DRIVER_SHA!r}}}))
'''


def package(nonce, phase, pins, route, pattern):
    source = guest_source(nonce, phase, pins, route, pattern)
    compile(source, "main.py", "exec")
    raw = DRIVER.read_bytes()
    common.require(common.sha(raw) == DRIVER_SHA, "fixed original driver changed")
    return {"source_kind": "uiflow2-package", "entrypoint": "main.py", "files": {
        "main.py": {"encoding": "utf8", "data": source},
        "official_flash.py": {"encoding": "base64", "data": base64.b64encode(raw).decode("ascii")}}}


class RouteLane(common.Lane):
    def __init__(self, client, args, output):
        super().__init__(client, args, args.device, "session", output)
        self.mode = args.mode == "flash"
        self.report.update(category="actual-guest-flashlight-route-lifecycle", flash_mode=self.mode,
                           guest_scope="S3 guest UART -> mem32 GPIO controller MMIO -> actual Unit lifecycle",
                           live_browser="NOT RUN", model_completeness="partial")

    def create(self):
        health = self.client.json("health")
        worker = health["capacity"]["worker"]
        common.require(worker["pid"] == self.args.expected_service_pid and worker["generation"] == self.args.expected_generation,
                       "service identity differs from explicit readiness handoff")
        body = {"device_id": self.device, "runtime": "qemu", "program_mode": "uiflow-live", "ttl_seconds": 1800,
                "ready_timeout_seconds": 40, "units": [{"id": "lamp", "unit_id": "flashlight", "port_id": PORTS[self.device],
                                                         "initial": {"flash-mode": self.mode}}]}
        if self.args.base_firmware_id:
            body.update(base_firmware_id=self.args.base_firmware_id, expected_base_sha256=self.args.expected_base_sha256)
        common.save(self.output / "create-intent.json", body)
        self.owned["creation_outcome"] = "unknown"
        try:
            created = self.client.json("sessions", "POST", body, timeout=120)
        except common.HTTPFailure as error:
            if error.complete and 400 <= error.status < 500 and error.status != 408:
                self.owned["creation_outcome"] = "rejected"
            raise
        self.owned.update(session_id=created["session_id"], creation_outcome="confirmed")
        self.session = common.poll(lambda: self.client.json("sessions/" + self.sid),
                                   lambda row: row["state"] in ("running", "failed", "stopped"), 120)
        common.require(self.session["state"] == "running", "session not running")
        self.owned["pid"] = self.session["pid"]
        identity = common.runtime_identity(self.session["pid"])
        common.require(identity["sha256"] == self.args.expected_runtime_sha256, "actual runtime hash differs")
        common.require(self.session["firmware_sha256"] == self.args.expected_firmware_sha256, "saved effective firmware differs")
        common.require(self.session["worker"]["generation"] == self.args.expected_generation, "session generation differs")
        self.pins = self.session["units"][0]["pins"]
        self.report.update(session_id=self.sid, pid=self.session["pid"], firmware_sha256=self.session["firmware_sha256"],
                           worker=self.session["worker"], runtime=identity, pins=self.pins)

    def units(self):
        return self.client.json("sessions/" + self.sid + "/units")

    def capture_zip(self):
        super().capture_zip()
        if "last_frame_sha256" not in self.report:
            import zipfile
            with zipfile.ZipFile(self.output / "evidence-before-stop.zip") as archive:
                raw = archive.read("frames/current.png")
            self.report["last_frame_sha256"] = common.sha(raw)
            self.report["failure_frame_reference"] = (
                "Actual before-stop ZIP frame for failure finalization only; no accepted functional/stable frame observation exists.")

    def phase(self, phase, route="ledc73", pattern="roundtrip"):
        self.same_vm()
        before = self.units()["units"][0]["properties"]
        offset = self.serial()["next_offset"]
        revision = self.client.json("program-revisions", "POST", package(self.nonce, phase, self.pins, route, pattern))["revision"]
        previous = self.report["runs"][-1]["execution_id"] if self.report["runs"] else None
        receipt = self.client.json("sessions/" + self.sid + "/program", "POST",
                                   {"revision_id": revision["revision_id"], "expected_execution_id": previous, "timeout_seconds": 30}, timeout=120)["program"]
        self.report["runs"].append(receipt)
        done = common.poll(lambda: self.client.json("sessions/" + self.sid + "/program"),
                           lambda row: row["program"]["state"] in ("completed", "failed", "unknown"), 60)["program"]
        common.require(done["state"] == "completed", "guest execution did not complete")
        for name in ("execution_id", "revision_id", "pid", "firmware_sha256"):
            common.require(done[name] == receipt[name], "GET/POST execution identity differs")
        common.require(done["revision_id"] == revision["revision_id"] and done["pid"] == self.session["pid"]
                       and done["firmware_sha256"] == self.session["firmware_sha256"], "Run changed original VM or revision")
        serial = self.serial(offset)["data"]
        marker = re.search(re.escape("__M5_EXEC_" + receipt["execution_id"] + "__:START") + r"\r?\n", serial)
        common.require(marker is not None, "fresh execution START missing")
        records = []
        for line in serial[marker.end():].splitlines(keepends=True):
            if line.endswith("\n") and line.startswith(PREFIX):
                records.append(json.loads(line[len(PREFIX):]))
        common.require(len(records) == 1, "fresh complete guest record missing/ambiguous")
        guest = records[0]
        common.require(all(guest[name] == expected for name, expected in (("nonce", self.nonce), ("phase", phase),
                       ("route_name", route), ("pattern", pattern), ("driver_sha256", DRIVER_SHA))), "guest record identity differs")
        unit = self.units()["units"][0]
        common.require(unit["available"] is True and not unit.get("errors") and unit["source"] == "guest-qom", "Unit state unavailable")
        props = unit["properties"]
        raw_qom = {}
        for name in PROPERTIES:
            row = self.client.json("sessions/" + self.sid + "/qom?" + urllib.parse.urlencode({"path": "/machine/units/lamp", "property": name}))
            raw_qom[name] = row
            common.require(row["value"] == props[name], "stable raw-QOM/Unit projection differs")
        common.require(props["en-level"] is bool(guest["en"]) and props["gpio-route-supported"] is (guest["route_effective"] == 256), "guest/controller route or EN differs")
        common.require(props["flash-mode"] is self.mode and props["mode-frozen"] is True and props["pad-synchronized"] is True, "fixed mode/pad synchronization differs")
        raw, headers = self.client.request("sessions/" + self.sid + "/frame.png?max_age_ms=1", binary=True)
        common.require(raw and headers.get("x-frame-sha256") == common.sha(raw) and headers.get("x-frame-execution-id") == done["execution_id"]
                       and headers.get("x-frame-revision-id") == done["revision_id"], "fresh frame identity differs")
        from PIL import Image
        with Image.open(io.BytesIO(raw)) as image:
            image.load()
            center = list(image.convert("RGB").getpixel((image.width // 2, image.height // 2)))
        common.require(center == ([255, 255, 255] if guest["en"] else [0, 0, 0]), "actual guest EN framebuffer pixel differs")
        label = f"{len(self.report['observations']):02d}-{route}-{pattern}-{phase}"
        (self.output / (label + ".png")).write_bytes(raw)
        self.report["last_frame_sha256"] = common.sha(raw)
        observation = {"phase": phase, "route_name": route, "pattern": pattern, "before_properties": before,
                       "execution": done, "guest": guest, "unit": unit, "raw_qom": raw_qom, "frame_headers": headers,
                       "frame_sha256": common.sha(raw), "selected_pad_pixel": center,
                       "snapshot_semantics": "separate stable-stage samples, never atomic across fields/frame"}
        common.save(self.output / (label + ".json"), observation)
        self.report["observations"].append(observation)
        return observation

    @staticmethod
    def unknown(row):
        p = row["unit"]["properties"]
        common.require(p["output-known"] is False and p["active"] is False and p["waveform-qualified"] is False,
                       "unsupported lifecycle incorrectly retained known output")
        common.require(p["program-state"] == "unqualified" and p["settle-at-ns"] == -1 and p["timeout-at-ns"] == -1,
                       "getter mask is insufficient: lifecycle/timers were not invalidated")

    @staticmethod
    def recovered_reset(row):
        p = row["unit"]["properties"]
        common.require(p["output-known"] is True and p["active"] is False and p["en-level"] is False
                       and p["program-state"] == "ready-low", "new supported EN fall/long-low did not recover known off")

    def web_hold(self):
        common.require(self.device == "cores3" and not self.mode, "mounted hold is CoreS3 Torch only")
        hold = self.args.hold_file
        paths = {name: hold.with_suffix(suffix) for name, suffix in
                 (("command", ".command.json"), ("observation", ".observation.json"), ("done", ".done"))}
        common.require(not hold.exists() and all(not path.exists() for path in paths.values()), "fresh owned hold required")
        hold.parent.mkdir(parents=True, exist_ok=True)
        common.save(hold, {"session_id": self.sid, "device_id": self.device, "mode": self.mode, "pid": self.session["pid"],
                          "firmware_sha256": self.session["firmware_sha256"], "worker": self.session["worker"],
                          "runtime_sha256": self.args.expected_runtime_sha256, "pins": self.pins, "nonce": self.nonce,
                          "output": str(self.output), "initial_observation": self.report["observations"][-1]})
        deadline = time.monotonic() + self.args.hold_seconds
        sequence = 0
        while time.monotonic() < deadline and not paths["done"].exists():
            if paths["command"].exists():
                command = json.loads(paths["command"].read_text())
                n, phase, route = command.get("sequence"), command.get("phase"), command.get("route_name")
                common.require(type(n) is int and n > 0 and phase in ("web-away", "web-return", "recover-reset", "recover-driver")
                               and route in ROUTES, "invalid owned mounted-observation command")
                if n > sequence:
                    common.require(n == sequence + 1, "mounted command sequence skipped")
                    row = self.phase(phase, route)
                    p, before = row["unit"]["properties"], row["before_properties"]
                    if phase in ("web-away", "web-return"):
                        self.unknown(row)
                        for name in ("edge-count", "rising-edges", "decoded-programs", "reset-events", "timeout-events"):
                            common.require(p[name] == before[name], "pure route operation generated edges/decode/reset/timeout")
                    elif phase == "recover-reset":
                        self.recovered_reset(row)
                    else:
                        common.require(p["output-known"] is True and p["active"] is True, "original-driver Torch recovery failed")
                    common.save(paths["observation"], {"sequence": n, "phase": phase, "route_name": route, "observation": row})
                    sequence = n
            time.sleep(0.2)
        common.require(paths["done"].exists(), "bounded mounted hold not acknowledged")
        done = json.loads(paths["done"].read_text())
        common.require(done.get("session_id") == self.sid and done.get("nonce") == self.nonce,
                       "mounted completion ownership differs")
        self.report["mounted_observer_ack"] = done
        # A failed observer must not be promoted to a mounted PASS. The API
        # result can be retained, but the combined requested run fails.
        common.require(done.get("observer_status") == "PASS", "mounted observer did not pass")

    def execute(self):
        try:
            self.create()
            self.recovered_reset(self.phase("bootstrap"))
            cases = [tuple(item.split(":")) for item in self.args.case] if self.args.case else CASES
            self.report["selected_cases"] = cases
            for route, pattern in ([] if self.args.web_only else cases):
                row = self.phase("atomic", route, pattern)
                self.unknown(row)
                p, before = row["unit"]["properties"], row["before_properties"]
                start_low = int(before["en-level"] is False)
                extra_fall = int(pattern != "roundtrip")
                extra_rise = int(pattern == "unsupported-low-rise")
                common.require(p["edge-count"] - before["edge-count"] == 2 + start_low + extra_fall + extra_rise
                               and p["rising-edges"] - before["rising-edges"] == 1 + start_low + extra_rise,
                               "route change fabricated pad edges or guest GPIO sequence differs")
                common.require(p["reset-events"] == before["reset-events"] + 1 and p["decoded-programs"] == before["decoded-programs"] + 1,
                               "prepared original program not decoded or unsupported low smuggled reset qualification")
                common.require(row["guest"]["events"][0]["arm_to_return_us"] < 100000, "route returned too late")
                stable = self.phase("stable-unknown", route, pattern)
                self.unknown(stable)
                for name in ("edge-count", "rising-edges", "decoded-programs", "reset-events", "timeout-events"):
                    common.require(stable["unit"]["properties"][name] == p[name], "old timer or unsupported low revived qualification")
                reset = self.phase("recover-reset", route, pattern)
                self.recovered_reset(reset)
                common.require(reset["unit"]["properties"]["reset-events"] == p["reset-events"] + 1, "new reset was not independently observed")
                restored = self.phase("recover-driver", route, pattern)
                rp = restored["unit"]["properties"]
                common.require(rp["output-known"] is True and rp["decoded-programs"] == reset["unit"]["properties"]["decoded-programs"] + 1,
                               "fixed original driver failed to recover decode")
                if self.mode:
                    common.require(rp["program-state"] in ("flash-active", "flash-expired") and rp["last-program-count"] == 1,
                                   "recovered single-pulse Flash code differs")
                else:
                    common.require(rp["active"] is True and rp["program-state"] == "torch-active", "recovered Torch not active")
            if self.args.web_only:
                row = self.phase("recover-driver")
                common.require(row["unit"]["properties"]["output-known"] is True and row["unit"]["properties"]["active"] is True,
                               "initial mounted Torch baseline is not active")
            if self.args.hold_file:
                self.web_hold()
            self.same_vm()
            self.capture_zip()
            self.report["status"] = "PASS"
        except BaseException as error:
            self.report["error"] = {"type": type(error).__name__, "message": str(error)}
        finally:
            self.report["cleanup"] = self.cleanup()
            if self.report["cleanup"]["status"] != "PASS":
                self.report["status"] = "FAIL"
            common.save(self.output / "report.json", self.report)
        return self.report


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--execute", action="store_true")
    parser.add_argument("--device", choices=PORTS, default="cores3")
    parser.add_argument("--mode", choices=("flash", "torch"), default="torch")
    parser.add_argument("--expected-runtime-sha256")
    parser.add_argument("--expected-firmware-sha256")
    parser.add_argument("--expected-generation", type=int)
    parser.add_argument("--expected-service-pid", type=int)
    parser.add_argument("--base-firmware-id")
    parser.add_argument("--expected-base-sha256")
    parser.add_argument("--out-dir", type=Path)
    parser.add_argument("--hold-file", type=Path)
    parser.add_argument("--hold-seconds", type=int, default=900)
    parser.add_argument("--web-only", action="store_true")
    parser.add_argument("--case", action="append", choices=[route + ":" + pattern for route, pattern in CASES])
    args = parser.parse_args()
    if not args.execute:
        print(json.dumps({"status": "NOT RUN", "api_contacted": False, "guests_started": 0, "hardware": "NOT RUN",
                          "devices": list(PORTS), "mode": args.mode, "cases": CASES, "source_entrypoint": "main.py",
                          "driver_sha256": DRIVER_SHA, "runtime_sha256": "REQUIRED FROM NEW RUNTIME READY HANDOFF",
                          "lifecycle": ["bootstrap", "atomic", "stable-unknown", "recover-reset", "recover-driver"],
                          "companion_status": "Pin/PWM generators prepared; not part of lifecycle PASS and not executed",
                          "web_status": "separate mounted observation plan; no browser claim"}, ensure_ascii=False, indent=2))
        return 0
    for name in ("expected_runtime_sha256", "expected_firmware_sha256"):
        common.require(re.fullmatch("[0-9a-f]{64}", getattr(args, name) or ""), "explicit normalized hash required")
    common.require(args.expected_generation and args.expected_generation > 0 and args.expected_service_pid and args.expected_service_pid > 0,
                   "explicit service readiness identity required")
    common.require(bool(args.base_firmware_id) == bool(args.expected_base_sha256), "explicit base pair required together")
    if args.expected_base_sha256:
        common.require(re.fullmatch("[0-9a-f]{64}", args.expected_base_sha256), "invalid explicit normalized base hash")
    common.require(1 <= args.hold_seconds <= 1800, "hold bound must be 1..1800 seconds")
    if args.hold_file or args.web_only:
        common.require(args.hold_file and args.device == "cores3" and args.mode == "torch", "mounted hold requires CoreS3 Torch and hold path")
    common.require(args.out_dir and not args.out_dir.exists(), "fresh output directory required")
    args.out_dir.mkdir(parents=True)
    (args.out_dir / "runner.executed.py.txt").write_bytes(Path(__file__).read_bytes())
    common.save(args.out_dir / "source-receipt.json", {"captured_before_api": True, "sha256": common.sha(Path(__file__).read_bytes()),
                "dependencies": {name: common.sha((ROOT / name).read_bytes()) for name in
                                 ("tests/units/run_binary_checkpoint_admission.py", "platform/tools/device_service.py",
