#!/usr/bin/env python3
"""Archive frozen owned evidence. No HTTP, guest, service or Git operation."""
import base64
from collections import Counter
import hashlib
import json
from pathlib import Path
import re
import zipfile

ROOT = Path.cwd()
ACTUAL = ROOT / ".work/flashlight-route-live-gen47-20261027"
PREPARED = ROOT / ".work/flashlight-route-live-preparation-20261027-r1"
DEST = ROOT / "docs/evidence/2026-10-27-flashlight-route-history/live"
OUT = ROOT / ".work/unit-host38-live-archive-20261027-r2"
FILES, WINDOWS, REDACTIONS = [], [], []
PNG = b"\x89PNG\r\n\x1a\n"


def sha(data):
    return hashlib.sha256(data).hexdigest()


def save(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + "\n", encoding="utf8")


def redacted(value, pointer=""):
    changed = []
    if isinstance(value, dict):
        value = dict(value)
        if value.get("encoding") == "base64" and isinstance(value.get("data"), str):
            b = base64.b64decode(value["data"], validate=True)
            value["data"] = {"redacted": True, "original_decoded_bytes": len(b), "original_decoded_sha256": sha(b),
                             "reason": "Original file-payload bytes excluded; exact original file is fingerprinted separately."}
            changed.append(pointer + "/data")
        for key, child in value.items():
            value[key], more = redacted(child, pointer + "/" + str(key))
            changed += more
    elif isinstance(value, list):
        result = []
        for i, child in enumerate(value):
            child, more = redacted(child, pointer + "/" + str(i))
            result.append(child)
            changed += more
        value = result
    return value, changed


def write(relative, raw, mode, source=None, original=None):
    path = DEST / relative
    assert not path.exists(), relative
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_bytes(raw)
    row = {"archive": relative, "mode": mode, "archive_bytes": len(raw), "archive_sha256": sha(raw)}
    if source is not None:
        row.update(source=str(source.relative_to(ROOT)), original_bytes=len(original), original_sha256=sha(original))
    FILES.append(row)


def zip_metadata(path, relative, raw):
    members = []
    with zipfile.ZipFile(path) as z:
        assert z.testzip() is None
        assert len(z.namelist()) == len(set(z.namelist()))
        for info in z.infolist():
            b = z.read(info.filename)
            members.append({"name": info.filename, "bytes": len(b), "sha256": sha(b), "published": False})
        serial = z.read("serial.log")
        pattern = re.compile(rb"__M5_EXEC_([0-9a-f]{32})__:START")
        for start in pattern.finditer(serial):
            execution = start.group(1)
            terminal = re.search(rb"__M5_EXEC_" + execution + rb"__:(?:DONE|ERROR)", serial[start.end():])
            assert terminal is not None
            end = start.end() + terminal.end()
            if serial[end:end + 2] == b"\r\n":
                end += 2
            block = serial[start.start():end]
            block.decode("ascii")
            window_path = relative.removesuffix(".zip") + ".serial-windows/" + execution.decode("ascii") + ".txt"
            write(window_path, block, "extracted-original-byte-window")
            WINDOWS.append({"archive": window_path, "source_zip": str(path.relative_to(ROOT)), "source_member": "serial.log",
                            "source_member_bytes": len(serial), "source_member_sha256": sha(serial), "start": start.start(),
                            "end": end, "window_sha256": sha(block), "execution_id": execution.decode("ascii"),
                            "scope": "Exact ASCII subsequence of original member bytes, not a transcoded full serial log."})
    meta = {"category": "HASH_ONLY_ORIGINAL_ZIP_AND_MEMBERS", "source": str(path.relative_to(ROOT)), "bytes": len(raw),
            "sha256": sha(raw), "original_binary_published": False, "members": members,
            "serial_policy": "Raw UART protocol bytes remain excluded. Exact original ASCII execution windows are separately indexed."}
    write(relative + ".metadata.json", (json.dumps(meta, indent=2) + "\n").encode(), "hash-only", path, raw)


def archive_file(path, group, base):
    assert not path.is_symlink()
    raw = path.read_bytes()
    relative = "artifacts/" + group + "/" + path.relative_to(base).as_posix()
    if path.suffix == ".zip":
        zip_metadata(path, relative, raw)
        return
    if raw.startswith((b"PK\x03\x04", b"PK\x05\x06", b"\x7fELF")):
        # HTTP evidence responses retain .response names. Their actual bytes
        # still make them excluded containers; never publish them as text.
        meta = {"original_binary_published": False, "bytes": len(raw), "sha256": sha(raw),
                "reason": "Container identified by original body magic; fingerprint only, regardless of filename suffix."}
        write(relative + ".metadata.json", (json.dumps(meta, indent=2) + "\n").encode(), "hash-only", path, raw)
        return
    if path.suffix in (".bin", ".rom", ".mpy", ".elf", ".uf2", ".img", ".pyc"):
        write(relative + ".metadata.json", (json.dumps({"original_binary_published": False, "bytes": len(raw), "sha256": sha(raw)}) + "\n").encode(), "hash-only", path, raw)
        return
    if path.suffix in (".py", ".mjs", ".md"):
        relative += ".txt"
    if not raw.startswith(PNG):
        try:
            raw.decode("utf8")
        except UnicodeDecodeError:
            meta = {"original_binary_published": False, "bytes": len(raw), "sha256": sha(raw),
                    "reason": "Non-UTF8, non-PNG body excluded; original bytes fingerprinted without transcoding."}
            write(relative + ".metadata.json", (json.dumps(meta, indent=2) + "\n").encode(), "hash-only", path, raw)
            return
    try:
        value = json.loads(raw)
    except (ValueError, UnicodeDecodeError):
        value = None
    if value is not None:
        value, changed = redacted(value)
        if changed:
            output = (json.dumps(value, ensure_ascii=False, indent=2) + "\n").encode("utf8")
            write(relative, output, "redacted-json", path, raw)
            REDACTIONS.append({"archive": relative, "json_pointers": changed, "original_sha256": sha(raw), "archive_sha256": sha(output)})
            return
    write(relative, raw, "original", path, raw)
