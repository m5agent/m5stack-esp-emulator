function rememberOwned(state, receipt) {
  assert.match(receipt.preview_id || '', /^[0-9a-f]{32}$/);
  if (state.previewId) assert.equal(receipt.preview_id, state.previewId);
  state.previewId = receipt.preview_id;
  assert.match(receipt.owner_epoch || '', /^[0-9a-f]{32}$/);
  if (state.ownerEpoch) assert.equal(receipt.owner_epoch, state.ownerEpoch);
  state.ownerEpoch = receipt.owner_epoch;
  if (receipt.session_id) {
    if (state.sessionId) assert.equal(receipt.session_id, state.sessionId);
    state.sessionId = receipt.session_id;
  }
}

// Only the original allocation intent can be replayed, with its exact key/body,
// to recover an unknown response for cleanup. No Run or input is replayed.
export async function createOwnedPreview(state, intent, send) {
  state.creation = {intent: structuredClone(intent), outcome: 'unknown'};
  try {
    const receipt = await send('previews', 'POST', state.creation.intent.body, state.creation.intent.key);
    rememberOwned(state, receipt); state.creation.outcome = 'confirmed'; return receipt;
  } catch (error) {
    if (error.responseComplete && error.status >= 400 && error.status < 500 && error.status !== 408) state.creation.outcome = 'rejected';
    throw error;
  }
}

export async function closeOwnedPreview(state, {send, wait, captureZip, pidGone, historyFrame, verify}) {
  state.cleanupWarnings ||= [];
  if ((!state.previewId || !state.ownerEpoch) && state.creation?.outcome === 'unknown') {
    const {body, key} = state.creation.intent;
    rememberOwned(state, await send('previews', 'POST', body, key));
    state.creation.recovered_for_cleanup = true;
  }
  if (!state.previewId) return null;
  const warn = (step, error) => state.cleanupWarnings.push({step, error: String(error)});
  try { rememberOwned(state, await send(`previews/${state.previewId}`)); }
  catch (error) { warn('preview-read', error); }
  assert.ok(state.ownerEpoch, 'Owned Preview epoch unavailable');
  if (state.sessionId && !state.session) {
    try { state.session = await send(`sessions/${state.sessionId}`); }
    catch (error) { warn('session-read', error); }
  }
  if (state.sessionId && !state.zipBeforeClose) {
    try { await captureZip(state.sessionId); state.zipBeforeClose = true; }
    catch (error) { warn('evidence-before-close', error); }
  }
  // A read, ZIP or lost close receipt must never skip the owned release.
  try { await send(`previews/${state.previewId}/close`, 'POST', {expected_owner_epoch: state.ownerEpoch}); }
  catch (error) { warn('close-receipt', error); }
  const closed = await wait(async () => {
    try { return await send(`previews/${state.previewId}`); }
    catch (error) { warn('close-observation', error); return null; }
  }, value => value?.state === 'closed', 120000);
  rememberOwned(state, closed);
  if (!state.sessionId) return {preview: closed, session_allocation: 'not published'};
  // Preserve observations before any optional frame/contract verification can
  // throw. Failed application/frame acceptance must not erase release proof.
  const value = {preview: closed, session_id: state.sessionId, zip_before_close: state.zipBeforeClose === true,
    last_frame_sha256: state.lastFrameSHA ?? null, full_frame_identity: 'NOT VERIFIED'};
  state.rawCleanup = value;
  try {
    value.session = await send(`sessions/${state.sessionId}`);
    value.capacity = await send('capacity');
    const pid = state.session?.pid ?? value.session.pid;
    value.pid_gone = Number.isInteger(pid) && pid > 0 ? await pidGone(pid) : false;
    value.pid_source = state.session ? 'before-close-owned-session' : 'retained-stopped-session';
    value.history_frame_sha256 = await historyFrame(state.sessionId);
    value.contract = verify(value);
    if (value.contract?.status === 'PASS') value.full_frame_identity = 'VERIFIED';
    return value;
  } catch (error) {
    value.verification_error = error.stack || String(error);
    value.contract = {status: 'FAIL', full_frame_identity: 'NOT VERIFIED'};
    error.rawCleanup = value;
    throw error;
  }
}

    return closeOwnedPreview(owned, {send: json, wait: until,
      captureZip: async id => { const zip = await json(`sessions/${id}/evidence`); await writeFile(resolve(output, 'failure-evidence.zip'), zip); },
      pidGone: pid => { try { process.kill(pid, 0); return false; } catch (error) { if (error.code === 'ESRCH') return true; throw error; } },
      historyFrame: async id => {
        const bytes = await json(`sessions/${id}/frame.png`);
        await writeFile(resolve(output, 'history-last-frame.png'), bytes); return sha(bytes);
      }, verify: value => pure('cleanup', value)});
  }
  try {
    const health = await json('health'); assert.equal(health.capacity.active_sessions, 0, 'Root-owned idle service required');
    const devices = await json('devices'); assert.ok(devices.devices.find(d => d.device_id === options.device)?.program_modes.includes('uiflow-live'));
    const inventory = await json(`units?device_id=${options.device}`); report.inventory_sha256 = sha(JSON.stringify(inventory));
    const createIntent = {key: randomUUID(), body: {device_id: options.device, runtime: 'qemu', ttl_seconds: 900, units: plan.attachments}};
    await save('create-intent.json', createIntent);
    const created = await createOwnedPreview(owned, createIntent, json);
    previewId = created.preview_id; ownerEpoch = created.owner_epoch; assert.match(previewId, /^[0-9a-f]{32}$/);
    const p = await until(() => json(`previews/${previewId}`), value => ['ready', 'failed'].includes(value.state), 120000);
    assert.equal(p.state, 'ready'); ownerEpoch = p.owner_epoch; sessionId = p.session_id;
    session = await json(`sessions/${sessionId}`); Object.assign(owned, {sessionId, session});
    report.runtime = runtimeIdentity(session.pid); assert.equal(report.runtime.sha256, expectedRuntime);
    Object.assign(report, {preview_id: previewId, owner_epoch: ownerEpoch, session_id: sessionId, pid: session.pid,
      firmware_sha256: session.firmware_sha256, worker: session.worker, frozen_attachments: session.units});
    assert.equal(session.units.length, 1); assert.equal(session.units[0].id, 'relay2'); assert.equal(session.units[0].unit_id, '2relay');
    assert.deepEqual(session.units[0].pins, plan.pins); assert.equal(session.units[0].port_id, plan.attachments[0].port_id);
    const {chromium} = await import('playwright');
    browser = await chromium.launch({executablePath: process.env.PLAYWRIGHT_CHROMIUM_EXECUTABLE || '/Applications/Google Chrome.app/Contents/MacOS/Google Chrome',
      headless: true, args: ['--use-gl=angle', '--use-angle=swiftshader', '--enable-unsafe-swiftshader']});
    page = await browser.newPage({viewport: {width: 1440, height: 1080}}); page.on('pageerror', error => report.page_errors.push(error.message));
    page.on('response', response => {
      if (!response.url().startsWith(base + '/api/v1/')) return;
      const index = report.browser_http.length, row = {path: new URL(response.url()).pathname, method: response.request().method(), status: response.status()};
      report.browser_http.push(row);
      if (row.method !== 'GET') row.body = response.request().postDataJSON();
      browserTasks.push(response.body().then(async bytes => {
        Object.assign(row, {bytes: bytes.length, sha256: sha(bytes), headers: await response.allHeaders()});
        await writeFile(resolve(output, `browser-${index}.response`), bytes);
      }).catch(error => { row.observation_error = error.message; }));
    });
    await page.route('**/platform/web/sessions.js*', async route => {
      const response = await route.fetch(), text = await response.text(); assert.equal(sha(text), report.source_sha256['platform/web/sessions.js']);
      await route.fulfill({response, body: text + '\nwindow.__relay2Preview={state,SessionVisual};\n'});
    });
    await page.goto(`${base}/platform/web/sessions.html`, {waitUntil: 'domcontentloaded'});
    await page.fill('#tokenInput', token); await page.click('#saveTokenBtn');
    const card = page.locator(`[data-session-id="${sessionId}"]`); await card.waitFor();
    await page.waitForFunction(id => window.__relay2Preview.state.visuals.get(id)?.unitPanel?.mounted.has('relay2'), sessionId);
    await apiRun(pure('seed', {nonce}), 'seed');
    for (const phase of ['00', '10', '01', '11']) {
      const run = await apiRun(packages[phase], phase), wanted = phase.split('').map(n => n === '1');
      const before = await observe(run, `fixed-${phase}`, wanted); await repeatStable(run, wanted, before); await frame(run, wanted, `fixed-${phase}`);
    }
    const interactive = await apiRun(packages.interactive, 'interactive');
    await rejectReverseInputs(interactive); await exerciseHost(interactive);
    await json(`previews/${previewId}/runs/${interactive.run_id}/cancel`, 'POST', {expected_owner_epoch: ownerEpoch});
    report.cancelled = await until(() => json(`previews/${previewId}/runs/${interactive.run_id}`), value => value.state === 'cancelled');
    assert.equal(report.cancelled.interrupt_ack, 'confirmed');
    const resumed = await apiRun(packages.interactive, 'interactive', 'interactive-after-cancel');
    await exerciseHost(resumed); await sameVM();
    for (const width of [390, 768, 1024, 1440]) {
      await page.setViewportSize({width, height: 1080}); await page.evaluate(() => scrollTo(0, 0)); await delay(200);
      assert.equal(await page.evaluate(() => document.documentElement.scrollWidth <= innerWidth), true);
      await page.screenshot({path: resolve(output, `relay2-${width}.png`), fullPage: true});
    }
    const zip = await json(`sessions/${sessionId}/evidence`); await writeFile(resolve(output, 'evidence-before-close.zip'), zip); owned.zipBeforeClose = true;
    report.zip = pure('zip', {bytes: zip.toString('base64'), expected: {session_id: sessionId, device_id: options.device,
      firmware_sha256: session.firmware_sha256, pid: session.pid, runs: report.runs, packages: report.packages}});
    // Freeze the same current Run's last stable image immediately before close.
    await frame(resumed, [false, false], 'before-close');
    report.cleanup = await closeOwned(); assert.equal(report.cleanup.preview.close_evidence.status, 'captured');
    assert.equal(report.cleanup.contract.status, 'PASS');
    await Promise.all(browserTasks); assert.deepEqual(report.page_errors, []);
    assert.deepEqual(report.browser_http.filter(row => row.method !== 'GET' && /\/sessions\/[^/]+\/(program|input|unit-input|renew|reset)$/.test(row.path)), []);
    assert.deepEqual(await fingerprints(), report.source_sha256); report.status = 'PASS';
  } catch (error) { report.error = error.stack || String(error); }
  finally {
    if (!report.cleanup) { try { report.failure_cleanup = await closeOwned(); } catch (error) { report.cleanup_error = error.stack || String(error); } }
    report.creation = owned.creation; report.cleanup_warnings = owned.cleanupWarnings;
    report.raw_cleanup = owned.rawCleanup;
    await browser?.close(); await Promise.all(browserTasks); await save('report.json', report);
  }
  console.log(`LIVE_PREVIEW_RELAY2_${report.status} ${options.device}`);
  if (report.status !== 'PASS') process.exitCode = 1;
}
