    nonce = 'a' * 32
    execution = 'b' * 32
    device = 'cores3'
    phase = '10'

    def package(self):
        return probe.build_package(self.nonce, self.device, probe.PINS[self.device], self.phase)

    def sample(self, *, phase=None):
        phase = phase or self.phase
        payload = probe.build_package(self.nonce, self.device, probe.PINS[self.device], phase)
        cwd = probe.GUEST_BASE + '/' + probe.revision_id(payload)
        return {'kind': 'sample', 'nonce': self.nonce, 'phase': phase, 'device': self.device, 'pins': [1, 2],
                'cwd': cwd, 'driver_file': cwd + '/relay2_official.py', 'driver_sha256': probe.DRIVER_SHA256,
                'driver_size': probe.DRIVER_SIZE, 'resource_hex': probe.RESOURCE.hex(), 'sequence': 1, 'events': 0,
                'levels': [True, False], 'statuses': [True, False], 'pads': [1, 0], 'initial_pads': [0, 0], 'written': [1, 0]}

    def window(self, row=None, *, ready=True, execution=None):
        row = row or self.sample()
        rows = []
        if ready:
            rows.append({k: v for k, v in row.items() if k in ('nonce', 'phase', 'device', 'pins', 'cwd', 'driver_file', 'driver_sha256', 'driver_size', 'resource_hex')} | {'kind': 'ready'})
        rows.append(row)
        return '\x01\x04\x04__M5_EXEC_' + (execution or self.execution) + '__:START\r\n' + ''.join(probe.PREFIX + json.dumps(value) + '\r\n' for value in rows)

    def parse(self, text, **kwargs):
        return probe.parse_window(text, self.execution, self.nonce, self.phase, self.device, probe.revision_id(self.package()), **kwargs)

    def test_independent_counters_accept_single_channel_and_reject_repeat_cross_talk(self):
        before = {'output-1-transitions': 3, 'output-2-transitions': 7, 'output-transitions': 10}
        self.assertEqual(probe.counter_contract(before, before)['channel_delta'], [0, 0])
        after = {'output-1-transitions': 4, 'output-2-transitions': 7, 'output-transitions': 11}
        self.assertEqual(probe.counter_contract(before, after, changed=1)['channel_delta'], [1, 0])
        after = {'output-1-transitions': 3, 'output-2-transitions': 8, 'output-transitions': 11}
        self.assertEqual(probe.counter_contract(before, after, changed=2)['channel_delta'], [0, 1])
        for after, changed in [({'output-1-transitions': 4, 'output-2-transitions': 8, 'output-transitions': 12}, 1),
                               ({'output-1-transitions': 3, 'output-2-transitions': 7, 'output-transitions': 11}, None),
                               ({'output-1-transitions': True, 'output-2-transitions': 7, 'output-transitions': 8}, None),

    def frame(self):
        raw = b'\x89PNG\r\n\x1a\nstandalone-byte-integrity-fixture'
        expected = {'preview_id': 'a' * 32, 'run_id': 'b' * 32, 'execution_id': 'c' * 32, 'revision_id': 'd' * 64}
        headers = {'X-Preview-ID': expected['preview_id'], 'X-Run-ID': expected['run_id'], 'X-Frame-Execution-ID': expected['execution_id'],
                   'X-Frame-Revision-ID': expected['revision_id'], 'X-Frame-SHA256': probe.sha(raw)}
        return raw, headers, expected

    def test_each_frame_header_and_exact_png_bytes_are_required(self):
        raw, headers, expected = self.frame()
        self.assertEqual(probe.frame_contract(raw, headers, expected)['status'], 'PASS')
        for key in headers:
            changed = {**headers, key: 'wrong'}
            with self.subTest(key=key), self.assertRaises(ValueError): probe.frame_contract(raw, changed, expected)
        with self.assertRaises(ValueError): probe.frame_contract(raw + b'changed', headers, expected)
        with self.assertRaises(ValueError): probe.frame_contract(b'not png', headers, expected)

    def test_frame_colors_are_exact_rgb565_endpoints_with_black_for_both_off(self):
        expected = [([False, False], [0, 0, 0], 0x000000), ([False, True], [0, 255, 0], 0x00FF00),
                    ([True, False], [255, 0, 0], 0xFF0000), ([True, True], [0, 0, 255], 0x0000FF)]
        for levels, rgb, color in expected:
            with self.subTest(levels=levels):
                self.assertEqual(probe.frame_color(levels), {'color_rgb888': color, 'rgb': rgb})
                self.assertTrue(all(component in (0, 255) for component in rgb))
        source = self.package()['files']['main.py']['data']
        self.assertIn('color = (0, 65280, 16711680, 255)[', source)
        self.assertNotIn('0x222222', source)
        self.assertNotIn('__FRAME_COLORS__', source)
        for levels in ([0, 0], [False], [False, False, False], '00'):
            with self.assertRaises(ValueError): probe.frame_color(levels)

    def archive(self, *, change_member=None, change_manifest=None):
        payload = self.package(); revision = probe.revision_id(payload); files = probe.decoded_files(payload)
        run = {'execution_id': self.execution, 'revision_id': revision}
        expected = {'session_id': 'owned', 'device_id': self.device, 'firmware_sha256': 'c' * 64, 'pid': 42,
                    'runs': [run], 'packages': {revision: payload}}
        prefix = 'programs/' + revision + '/'
        members = {f'programs/{self.execution}.json': probe.canonical({**run, 'pid': 42, 'firmware_sha256': 'c' * 64}),
                   'serial.log': b'actual serial', 'frames/current.png': b'PNG', 'qom/snapshot.json': b'{}',
                   'units/contract.json': b'{}', 'units/state.json': b'{}', 'units/runtime.json': b'{}',
                   prefix + 'source.json': probe.canonical({'entrypoint': 'main.py', 'files': {p: {'encoding': 'base64', 'data': base64.b64encode(v).decode()} for p, v in files.items()}}),
                   prefix + 'revision.json': probe.canonical({'revision_id': revision, 'source_kind': 'uiflow2-package', 'files': [{'path': p, 'size': len(v), 'sha256': probe.sha(v)} for p, v in files.items()]})}
        members.update({prefix + 'files/' + p: v for p, v in files.items()})
        if change_member: change_member(members, prefix)
        manifest = {'session': {k: expected[k] for k in ('session_id', 'device_id', 'firmware_sha256', 'pid')},
                    'files': {p: {'size': len(v), 'sha256': probe.sha(v)} for p, v in members.items()}}
        if change_manifest: change_manifest(manifest)
        output = io.BytesIO()
        with zipfile.ZipFile(output, 'w') as archive:
            for path, value in members.items(): archive.writestr(path, value)
            archive.writestr('manifest.json', json.dumps(manifest))
        return output.getvalue(), expected

    def test_zip_binds_vm_each_run_entire_original_package_and_runtime_evidence(self):
        data, expected = self.archive(); self.assertEqual(probe.inspect_zip(data, expected)['status'], 'PASS')
        with self.assertRaises(ValueError): probe.inspect_zip(data, {**expected, 'pid': 99})
        mutations = [lambda m, p: m.__setitem__(p + 'files/relay2_official.py', probe.OFFICIAL_DRIVER[:-1] + b' '),
                     lambda m, p: m.__setitem__(p + 'source.json', b'{}'),
                     lambda m, p: m.pop('qom/snapshot.json'),
                     lambda m, p: m.__setitem__('../escape', b'unsafe'),
                     lambda m, p: m.__setitem__('programs/' + self.execution + '.json', b'{}')]
        for mutation in mutations:
            with self.subTest(mutation=mutation), self.assertRaises((ValueError, KeyError)): probe.inspect_zip(*self.archive(change_member=mutation))
        with self.assertRaises(ValueError): probe.inspect_zip(*self.archive(change_manifest=lambda m: m['files']['serial.log'].update(sha256='0' * 64)))
