#!/usr/bin/env python3
"""Read-only validation of the distinct AtomS3R pure-parser IPC failed attempt."""
from collections import defaultdict
import hashlib
from importlib.machinery import SourceFileLoader
from importlib.util import module_from_spec, spec_from_loader
import io
import json
from pathlib import Path
import zipfile

from PIL import Image

HERE = Path(__file__).resolve().parents[1]


def sha(value):
    return hashlib.sha256(value).hexdigest()


def expect(value, reason):
    if not value:
        raise AssertionError(reason)


def key(value):
    return json.dumps(value, sort_keys=True, separators=(',', ':'), ensure_ascii=False)


def audit():
    folder = HERE / 'atoms3r-r1'
    report = json.loads((folder / 'raw/report.json').read_bytes())
    receipt = json.loads((folder / 'raw/root-pure-ipc-stall.json').read_bytes())
    provenance = json.loads((folder / 'provenance.json').read_bytes())
    source_refs = []
    helper_path = None
    for path, expected in receipt['source_sha256'].items():
        snapshot = HERE.parents[1] / 'sources' / (path.replace('/', '__') + '.txt')
        expect(sha(snapshot.read_bytes()) == expected == report['source_sha256'][path], 'shared captured source fingerprint')
        source_refs.append({'repository_path': path, 'snapshot': '../../sources/' + snapshot.name, 'sha256': expected,
                            'provenance': 'captured shared snapshot, original attempt fingerprint matched'})
        if path.endswith('/relay2_preview_probe.py'):
            helper_path = snapshot
    loader = SourceFileLoader('atoms3r_failed_helper', str(helper_path))
    spec = spec_from_loader(loader.name, loader)
    helper = module_from_spec(spec)
    loader.exec_module(helper)
    expect(report['status'] == 'FAIL' and report['device_id'] == 'atoms3r', 'distinct failed AtomS3R attempt')
    expect(len(report['runs']) == 6 and len(report['observations']) == 25, 'recorded execution boundary')
    expect([r['phase'] for r in report['runs']] == ['seed', '00', '10', '01', '11', 'interactive'], 'completed fixed cases before interactive failure')
    sample = (folder / 'parse-stall-sample.txt').read_bytes()
    sample_origin = provenance['files']['parse-stall-sample.txt']
    expect(receipt['sample_sha256'] == sample_origin['original_sha256'], 'root receipt binds original sampling bytes')
    expect(sha(sample) == sample_origin['sha256'], 'archived sampling bytes integrity')
    lookup = defaultdict(list)
    with zipfile.ZipFile(folder / 'raw-evidence.zip') as compressed:
        expect(compressed.testzip() is None and 'report.json' not in compressed.namelist(), 'compressed raw CRC/report exclusion')
        expect(len(compressed.namelist()) == len(set(compressed.namelist())), 'unique raw members')
        for index, row in enumerate(report['http']):
            value = compressed.read(f'http-{index}.response')
            expect(row['bytes'] == len(value) and row['sha256'] == sha(value), 'HTTP response bytes')
            try:
                lookup[key(json.loads(value))].append(index)
            except (ValueError, UnicodeError):
                pass
        saved = missing = 0
        for index, row in enumerate(report['browser_http']):
            name = f'browser-{index}.response'
            if name in compressed.namelist():
                value = compressed.read(name)
                expect(row['bytes'] == len(value) and row['sha256'] == sha(value), 'saved browser body bytes')
                saved += 1
            else:
                expect('observation_error' in row, 'explicit unsaved browser error')
                missing += 1
    runs = {run['run_id']: run for run in report['runs']}
    observations = frames = 0
    for run in runs.values():
        execution = run['last_observation']
        expect(execution['pid'] == report['pid'] and execution['firmware_sha256'] == report['firmware_sha256'], 'same VM recorded for six Runs')
    for item in report['observations']:
        if 'serial' in item:
            run = runs[item['run_id']]
            serial, fresh = item['serial'], item['serial']['fresh']
            expect(key(serial['complete']) in lookup, 'UART raw response')
            if fresh:
                expect(key(fresh) in lookup and fresh['offset'] >= run['serialOffset'], 'fresh UART raw response')
            parsed = helper.parse_window(serial['complete']['data'], run['execution_id'], report['nonce'], run['phase'], 'atoms3r', run['revision_id'],
                                         require_ready=True, fresh_text=fresh['data'] if fresh else None)
            expect(parsed == serial['parsed'] and parsed['sample']['levels'] == item['wanted'], 'typed original-driver UART evidence')
            for field, value in item['qom'].items():
                expect(key(value) in lookup and value['property'] == field and value['path'] == '/machine/units/relay2', 'raw independent GPIO QOM')
            for channel in (1, 2):
                led = item['mounted']['leds'][channel - 1]
                expect(item['qom'][f'output-{channel}-level']['value'] == item['wanted'][channel - 1], 'GPIO level agrees with guest')
                expect(led['attached'] is led['known'] is led['userDataKnown'] is True and led['channel'] == channel, 'mounted LED/channel')
                expect(led['state']['brightness'] == int(item['wanted'][channel - 1]), 'mounted LED level')
            observations += 1
        if item['label'].startswith('frame-'):
            run = runs[item['run_id']]
            value = (folder / 'raw' / (item['label'].removeprefix('frame-') + '-frame.png')).read_bytes()
            helper.frame_contract(value, item['headers'], {k: run[k] for k in ('preview_id', 'run_id', 'execution_id', 'revision_id')})
            frames += 1
    cleanup = report['failure_cleanup']
    helper.cleanup_contract(cleanup)
    for scope in ('preview', 'session', 'capacity'):
        expect(key(cleanup[scope]) in lookup, 'failure cleanup backed by raw HTTP')
    expect(cleanup['preview']['state'] == 'closed' and cleanup['session']['state'] == 'stopped' and cleanup['session']['flash_retained'] is False
           and cleanup['pid_gone'] is True and cleanup['capacity']['active_sessions'] == cleanup['capacity']['worker']['active_lease_count'] == 0, 'resource release after helper interruption')
    evidence = (folder / 'raw/failure-evidence.zip').read_bytes()
    zip_check = helper.inspect_zip(evidence, {**report, 'packages': report['packages']})
    pngs = []
    for path in sorted((folder / 'raw').glob('*.png')):
        with Image.open(path) as image:
            image.load()
            pngs.append({'path': path.name, 'width': image.width, 'height': image.height})
    interactive = [item['serial']['parsed']['sample'] for item in report['observations'] if 'serial' in item and item['run_id'] == report['runs'][-1]['run_id']]
    expect(interactive and all(item['events'] == 0 for item in interactive), 'no interactive event closure may be claimed')
    return {'attempt': 'atoms3r-r1', 'device_id': 'atoms3r', 'attempt_status': 'FAIL', 'automatic_report_status': report['status'], 'root_classification': 'FAIL',
            'formal_acceptance': False, 'failure_kind': 'synchronous pure-parser child IPC EOF stall',
            'evidence_level': 'fixed combinations and original-driver GPIO/LED observations; interactive event closure NOT VERIFIED; finally resource cleanup verified',
            'reasons': ['After six Runs the interactive loop was observed at events=0; the synchronous pure parse helper then blocked waiting for stdin EOF.',
                        'Root SIGTERM targeted the exact runner-owned parser helper, allowing the existing finally path to close normally.',
                        'This is distinct from the CoreS3 r2 unbounded browser-observer drain recovery and remains FAIL.'],
            'hardware': 'NOT RUN', 'runtime': report['runtime'], 'worker': report['worker'], 'preview_id': report['preview_id'], 'session_id': report['session_id'],
            'firmware_sha256': report['firmware_sha256'], 'sources': source_refs, 'sampling_original_sha256': receipt['sample_sha256'],
            'checks': {'runs': 6, 'observations': 25, 'typed_uart_qom_mounted_observations_verified': observations, 'five_header_frames_verified': frames,
                       'raw_http_responses_verified': len(report['http']), 'browser_response_bodies_verified': saved, 'explicit_unsaved_browser_body_errors': missing,
                       'zip': zip_check, 'pngs_decoded': pngs, 'interactive_events_verified': 0},
            'resource_cleanup': {'status': cleanup['contract']['status'], 'preview_state': cleanup['preview']['state'], 'session_state': cleanup['session']['state'],
                                 'flash_retained': cleanup['session']['flash_retained'], 'pid_gone_recorded': cleanup['pid_gone'], 'active_sessions': cleanup['capacity']['active_sessions'],
                                 'active_lease_count': cleanup['capacity']['worker']['active_lease_count']},
            'limits': ['No interactive-event success, physical-hardware result or later-version acceptance is inferred.',
                       'This offline audit verifies recorded resource/PID evidence and does not query a live service or current process table.']}


if __name__ == '__main__':
    print(json.dumps(audit(), ensure_ascii=False, indent=2))
