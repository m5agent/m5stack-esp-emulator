def errors(schema, value):
    wrapped = {"$schema": "https://json-schema.org/draft/2020-12/schema",
               "components": document["components"], "allOf": [schema]}
    jsonschema.Draft202012Validator.check_schema(wrapped)
    validator = jsonschema.Draft202012Validator(wrapped, format_checker=jsonschema.FormatChecker())
    return [{"path": list(error.path), "message": error.message} for error in validator.iter_errors(value)]

def request(name, method, template, path, payload=None, *, key=None, headers=None, token=True, raw=None, contextual=False):
    operation = document["paths"][template][method.lower()]
    request_headers = {"Content-Type": "application/json", **(headers or {})}
    if token:
        request_headers["Authorization"] = "Bearer synthetic-host-contract-control"
    if key is not None:
        request_headers["Idempotency-Key"] = key
    encoded = json.dumps(payload).encode() if payload is not None else raw
    connection = http.client.HTTPConnection("127.0.0.1", server.server_port, timeout=3)
    try:
        connection.request(method, "/api/v1" + path, body=encoded, headers=request_headers)
        response = connection.getresponse()
        body = response.read()
        status, received_headers = response.status, dict(response.getheaders())
        result = json.loads(body) if "json" in received_headers.get("Content-Type", "") else None
    finally:
        connection.close()
    record = {"name": name, "method": method, "template": template, "status": status,
              "response": result, "body_sha256": hashlib.sha256(body).hexdigest()}
    if payload is not None:
        request_schema = operation["requestBody"]["content"]["application/json"]["schema"]
        failures = errors(request_schema, payload)
        row = {"name": name, "schema_valid": not failures, "http_accepted": 200 <= status < 300,
               "contextual_rejection": contextual, "errors": failures}
        if len(encoded) < 10000:
            record["request"] = payload
        report["request_checks"].append(row)
        if not contextual and row["schema_valid"] != row["http_accepted"]:
            report["shape_mismatches"].append(row)
    if str(status) not in operation["responses"]:
        report["undocumented_statuses"].append({"name": name, "template": template, "method": method, "status": status})
    declared = operation["responses"].get(str(status), {})
    if "$ref" in declared:
        declared = document["components"]["responses"][declared["$ref"].rsplit("/", 1)[1]]
    schema = declared.get("content", {}).get("application/json", {}).get("schema")
    if schema is not None:
        report["response_checks"].append({"name": name, "status": status, "errors": errors(schema, result)})
    (OUT / (name + ".json")).write_text(json.dumps(record, indent=2) + "\n")
    return status, result, received_headers, body
