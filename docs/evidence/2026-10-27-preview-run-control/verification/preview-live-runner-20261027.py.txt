def bad(path,payload,code,method='POST',key=None,status=409):
 try:call(path,method,payload,key)
 except RuntimeError:

def verify_zip(data):
 with zipfile.ZipFile(io.BytesIO(data)) as archive:
  names=archive.namelist();require(len(names)==len(set(names)) and archive.testzip() is None,'corrupt/duplicate ZIP')
  for name in names:
   path=PurePosixPath(name);require(not path.is_absolute() and '..' not in path.parts and chr(92) not in name,'unsafe ZIP path')
   require((archive.getinfo(name).external_attr>>16)&0xf000!=0xa000,'ZIP symlink')
  manifest=json.loads(archive.read('manifest.json'))
  require(set(manifest['files'])==set(names)-{'manifest.json'},'ZIP index differs')
  for name,meta in manifest['files'].items():
   content=archive.read(name);require(meta['sha256']==sha(content) and meta['size']==len(content),'ZIP hash/size differs')
  ident=manifest['session'];require(ident['state']=='running' and ident['session_id']==sid and ident['pid']==pid and ident['firmware_sha256']==report['firmware_sha256'],'ZIP session identity differs')
  require({'serial.log','frames/current.png','qom/snapshot.json','units/contract.json','units/state.json','units/runtime.json'}<=set(names),'ZIP required data plane missing')
  props=live_unit(manifest['units'],sid,args.unit);require(int(props['input-level' if args.unit=='pir' else 'output-level'])==0,'ZIP final Unit differs')
  require(live_unit(json.loads(archive.read('units/state.json')),sid,args.unit)==props,'ZIP Unit state/manifest differ')
  frame_observation(archive.read('frames/current.png'),(320,240),'C')
  raw=archive.read('serial.log');serial_text=raw.decode('utf-8',errors='replace')
  require(serial_text.count('ESP-ROM:')==report['boot_rom_count'] and serial_text.count('M5EMU_UART_REPL_READY')==1,'ZIP reboot marker differs')
  for row in observations:
   marker=('__M5_EXEC_'+row['execution_id']+'__:START').encode()
   require(raw.find(marker)>=0 and raw.find(marker)+len(marker)<=row['offset'],'ZIP UART window precedes expected execution START')
   part=raw[row['offset']:row['window']['next_offset']].decode('utf-8',errors='replace')
   require(part==row['window']['data'] and matching_json_line(part,'LIVE_PACKAGE_SAMPLE ',row['expected']) is not None,'ZIP fresh UART binding differs')
  stored={PurePosixPath(n).stem for n in names if n.startswith('programs/') and n.count('/')==1 and n.endswith('.json')}
  expected={run['execution_id']:run for run in report['runs'].values()}
  require(stored<=set(expected),'ZIP unexpected execution: retry may have executed again')
  require({report['runs'][key]['execution_id'] for key in ('finite','A','burst-D','after-cancel-C')}<=stored,'ZIP missing required execution')
  for execution in stored:
   run=expected[execution];history=json.loads(archive.read('programs/'+execution+'.json'))
   require(history['execution_id']==execution and history['revision_id']==run['revision_id'] and history['pid']==pid and history['firmware_sha256']==report['firmware_sha256'],'ZIP execution identity differs')
   if execution==report['cancelled']['execution_id']:require(history['state']=='interrupted' and history['interruption_confirmed'] is True,'ZIP cancelled execution not confirmed')
   version=next(k for k,v in checkpoints.items() if v==run['revision_id']);files=packages[version]
   deployment=history.get('deployment',{})
   if history.get('acknowledged'):
    require(deployment.get('phase')=='activated' and deployment.get('inventory')==file_inventory(files) and deployment.get('package_path')=='/flash/.m5-live-packages/'+run['revision_id'],'ZIP guest deployment inventory differs')
   meta=json.loads(archive.read('programs/'+run['revision_id']+'/revision.json'))
   require(meta==checkpoint_metadata[version],'ZIP checkpoint original metadata differs')
   require({r['path']:{'size':r['size'],'sha256':r['sha256']} for r in meta['files']}==file_inventory(files),'ZIP inventory differs')
   for name,content in files.items():require(archive.read('programs/'+run['revision_id']+'/files/'+name)==content,'ZIP original file differs')
   require(revision_id(files)==run['revision_id'],'ZIP full package ID differs')
  return {'members':len(names),'sha256':sha(data),'execution_count':len(stored),'member_hashes_original_files_serial_identity_checked':True}
