import hashlib,json,re,zipfile
from pathlib import Path
root=Path.cwd();dest=root/'docs/evidence/2026-10-27-preview-run-control'
patterns={
 'provider_key':rb'\b(?:sk-(?:proj-)?[A-Za-z0-9_-]{20,}|gh[pousr]_[A-Za-z0-9]{30,}|github_pat_[A-Za-z0-9_]{30,})\b',
 'private_key':rb'-----BEGIN (?:RSA |EC |OPENSSH )?PRIVATE KEY-----',

}
hits=[];files=members=0
for path in sorted(p for p in dest.rglob('*') if p.is_file()):
 files+=1
 data=path.read_bytes()

 if path.suffix=='.zip':
  with zipfile.ZipFile(path) as z:
   assert z.testzip() is None
   for entry in z.namelist():
    members+=1;assert not entry.lower().endswith(('.bin','.rom','.elf')),entry
    for name,pattern in patterns.items():
     if re.search(pattern,z.read(entry)):hits.append({'file':str(path.relative_to(dest))+'!'+entry,'rule':name})
token=(root/'.work/preview-control-token').read_bytes().strip()
for p in dest.rglob('*'):
 if p.is_file():

   with zipfile.ZipFile(p) as z:
