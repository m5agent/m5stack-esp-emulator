PIXELS = b"HOST_ONLY_NOT_GUEST_IDENTICAL_FRAME_BYTES"


class _Block:
    def __init__(self):
        self.entered, self.release = threading.Event(), threading.Event()

    def wait(self):
        self.entered.set()
        if not self.release.wait(3):
            raise AssertionError("host fixture blocking point was not released")


class _Session:
    def __init__(self, service, binding, request):
        self.service = service
        self.id, self.pid = uuid.uuid4().hex, 42424242
        self.worker_generation = 44
        self.process_start_token = "host-only-process-start"
        self._program_capabilities = {}
        self.firmware_sha = "d" * 64
        self.firmware_id = "e" * 64
        self.state = "running"
        self.program_transport = "uart-repl"
        self.management_binding = deepcopy(binding)
        self.handle = object()
        self.registry = UnitRegistry()
        self.units = self.registry.attachments("cores3", "qemu", request.get("units", []))
        self.unit_contract = self.registry.freeze([unit["unit_id"] for unit in self.units]) if self.units else None
        self.spec, self.runtime = service.specs["cores3"], "qemu"
        self.program = None
        self.run_calls, self.interrupt_calls, self.inputs, self.events = [], [], [], []
        self.run_hook = self.interrupt_hook = self.input_hook = self.frame_hook = self.stop_hook = None
        self.dir = service.state_dir / "sessions" / self.id
        self.dir.mkdir(parents=True)
        self.last_frame_available = False
        self.ttl_seconds = request["ttl_seconds"]

    def require_management_handle(self, handle=None):
        if handle is not self.handle:
            raise ServiceError("session_managed_by_preview", "host fixture capability mismatch", 409)

    def snapshot(self):
        return {"session_id": self.id, "state": self.state, "pid": self.pid, "device_id": "cores3",
                "runtime": self.runtime, "firmware_id": self.firmware_id, "firmware_sha256": self.firmware_sha,
                "program_transport": self.program_transport, "program": deepcopy(self.program),
                "worker": {"generation": self.worker_generation}, "program_capabilities": self.program_capabilities(),
                "units": deepcopy(self.units), "unit_contract": deepcopy(self.unit_contract), "management_binding": deepcopy(self.management_binding),
                "flash_retained": self.state != "stopped", "last_frame_available": self.last_frame_available}

    def program_status(self):
        return {"session_id": self.id, "program_transport": self.program_transport, "program": deepcopy(self.program)}

    _program_identity = DeviceSession._program_identity
    program_capabilities = DeviceSession.program_capabilities

    def _owner(self):
        return self.program["execution_id"] if self.program else None

    def run_revision(self, revision_id, *, execution_id, expected_execution_id, timeout_seconds=120,
                     cancelled=None, on_synchronized=None, management_handle=None):
        self.require_management_handle(management_handle)
        if expected_execution_id != self._owner():
            raise ServiceError("program_generation_conflict", "host fixture owner mismatch", 409)
        call = {"revision_id": revision_id, "execution_id": execution_id,
                "expected_execution_id": expected_execution_id, "timeout_seconds": timeout_seconds,
                "cancelled": cancelled, "on_synchronized": on_synchronized}
        self.run_calls.append(call)
        self.events.append(("run-entered", execution_id))
        self.program = {"execution_id": execution_id, "revision_id": revision_id, "state": "submitting",
                        "pid": self.pid, "firmware_sha256": self.firmware_sha, "serial_offset": 0}
        if on_synchronized:
            on_synchronized()
        result = self.run_hook(call) if self.run_hook else {"state": "running", "acknowledged": True}
        self.program.update(result)
        self.events.append(("run-returned", execution_id))
        return self.program_status()

    def interrupt_program(self, *, expected_execution_id, timeout_seconds=10, management_handle=None):
        self.require_management_handle(management_handle)
        if expected_execution_id != self._owner():
            raise ServiceError("program_generation_conflict", "host fixture interrupt owner mismatch", 409)
        self.interrupt_calls.append({"expected_execution_id": expected_execution_id, "timeout_seconds": timeout_seconds})
        self.events.append(("interrupt-entered", expected_execution_id))
        ack = self.interrupt_hook() if self.interrupt_hook else {"interruption_confirmed": True, "serial_offset": 0, "prompt_end": 30}
        if not ack.get("interruption_confirmed"):
            raise ServiceError("program_timeout", "HOST_ONLY_NOT_GUEST prompt unconfirmed", 504)
        if self.program:
            self.program.update(state="interrupted", interruption_confirmed=True, interrupt_ack=ack)
        self.events.append(("interrupt-confirmed", expected_execution_id))
        return {**self.program_status(), "interrupt_ack": ack}

    def frame_metrics(self, *, analyze=False, **kwargs):
        if self.frame_hook:
            self.frame_hook()
        self.last_frame_available = True
        return {"png": PIXELS, "width": 2, "height": 2, "source": "HOST_ONLY_NOT_GUEST",
                "captured_at": datetime.now(timezone.utc).isoformat(), "captured_monotonic": time.monotonic(),
                "execution_id": self._owner(), "revision_id": self.program.get("revision_id") if self.program else None,
                "program_state": self.program.get("state") if self.program else None}

    def preview_frame(self, max_age_ms):
        return {**self.frame_metrics(), "sample_id": "host-fixture-sample", "sequence": 1,
                "sha256": hashlib.sha256(PIXELS).hexdigest(), "age_ms": 0, "cache_hit": False}

    def inject(self, payload):
        captured = self._owner()
        self.events.append(("input-entered", captured))
        if self.input_hook:
            self.input_hook()
        self.inputs.append((captured, deepcopy(payload)))
        self.events.append(("input-finished", self._owner()))
        return {"session_id": self.id, "accepted": True}

    def unit_input(self, payload):
        return self.inject(payload)

    def validate_input(self, payload):
        return DeviceSession.validate_input(self, payload)

    def validate_unit_input(self, payload):
        return DeviceSession.validate_unit_input(self, payload)

    def _unit_registry(self):
        return DeviceSession._unit_registry(self)

    def renew(self, ttl):
        self.ttl_seconds = ttl
        return self.snapshot()

    def evidence_bundle(self, include=None):
        self.events.append(("evidence", self._owner()))
        stream = io.BytesIO()
        with zipfile.ZipFile(stream, "w") as archive:
            archive.writestr("HOST_ONLY_NOT_GUEST.json", json.dumps(self.snapshot()))
        return stream.getvalue()

    def stop(self, purge=False, *, retain_flash=None):
        self.events.append(("stop", self._owner()))
        if self.stop_hook:
            result = self.stop_hook()
            if result is not None:
                return result
        self.state = "stopped"
        return self.snapshot()


class _Service:
    def __init__(self, state_dir):
        self.state_dir = state_dir
        self.specs = {"cores3": SimpleNamespace(id="cores3", supported=True, backend_ready=True, expected_size=(320, 240))}
        self.program_packages = ProgramPackageStore(state_dir / "packages")
        self.sessions, self.allocations = {}, []
        self._sessions = self.sessions
        self._lock = threading.RLock()
        self.allocation_hook = None
        self.reference_sha = "c" * 64
        self.base_calls = []

    def reference_info(self, spec):
        return {"sha256": self.reference_sha}

    def program_base_info(self, spec, *, base_firmware_id=None, expected_base_sha256=None):
        self.base_calls.append((spec.id, base_firmware_id, expected_base_sha256))
        if base_firmware_id != expected_base_sha256:
            raise ServiceError("uiflow_base_hash_mismatch", "host fixture base pin differs", 409)
        return {"sha256": base_firmware_id, "firmware_id": base_firmware_id, "kind": "stored-manifest"}

    def preview_allocation_guard(self):
        return nullcontext()

    def create_managed_session(self, payload, **binding):
        if binding["expected_reference_sha256"] != payload.get("expected_base_sha256", self.reference_sha):
            raise ServiceError("uiflow_base_sha256_mismatch", "host fixture reference changed", 409)
        if self.allocation_hook:
            self.allocation_hook(payload, binding)
        session = _Session(self, binding, payload)
        self.sessions[session.id] = session
        self.allocations.append((deepcopy(payload), deepcopy(binding)))
        return session.snapshot(), session.handle

    def get_session(self, session_id):
        if session_id not in self.sessions:
            raise ServiceError("unknown_session", "host fixture session absent", 404)
        return self.sessions[session_id]

    def recover_management_handle(self, session_id, **binding):
        session = self.get_session(session_id)
        if session.management_binding != binding:
            raise ServiceError("session_managed_by_preview", "host fixture binding mismatch", 409)
        return session.handle

    def delete_session(self, session_id, purge=False, *, retain_flash=None):
        return self.get_session(session_id).stop(purge=purge, retain_flash=retain_flash)

    def list_sessions(self, include_stopped=True):
        return {"sessions": [session.snapshot() for session in self.sessions.values()]}


class PreviewServiceTests(unittest.TestCase):
    def setUp(self):
        self.temp = tempfile.TemporaryDirectory(prefix="m5-preview-host-")
        self.addCleanup(self.temp.cleanup)
        self.service = _Service(Path(self.temp.name))
        self.control = PreviewService(self.service, _start_workers=False)
        self.controls, self.threads, self.blocks = [self.control], [], []
        self.addCleanup(self.finish)
        self.no_process = mock.patch("subprocess.Popen", side_effect=AssertionError("HOST_ONLY_NOT_GUEST: process launch forbidden"))
        self.no_process.start()
        self.addCleanup(self.no_process.stop)

    def finish(self):
        for block in self.blocks:
            block.release.set()
        for control in self.controls:
            control.shutdown()
        for thread in self.threads:
            thread.join(timeout=3)
            self.assertFalse(thread.is_alive(), "host fixture thread leaked")

    def block(self):
        block = _Block()
        self.blocks.append(block)
        return block

    def background(self, callback):
        result = {}
        def run():
            try:
                result["value"] = callback()
            except BaseException as error:
                result["error"] = error
        thread = threading.Thread(target=run, daemon=True)
        self.threads.append(thread)
        thread.start()
        return thread, result

    def create(self, key="create", **changes):
        return self.control.create({"device_id": "cores3", "units": [], "ttl_seconds": 300, **changes},
                                   owner=LOCAL_OWNER, idempotency_key=key)

    def package(self, version="A", **files):
        return self.service.program_packages.create({"files": {"main.py": {"encoding": "utf8", "data": f"print({version!r})\n"}, **files}})["revision_id"]

    def submit(self, preview_id, revision_id=None, key=None, **changes):
        preview = self.control.get(preview_id, owner=LOCAL_OWNER)
        payload = {"revision_id": revision_id or self.package(key or "A"), "expected_generation": preview["generation"],
                   "expected_owner_epoch": preview["owner_epoch"], "timeout_seconds": 120, **changes}
        return self.control.submit(preview_id, payload, owner=LOCAL_OWNER, idempotency_key=key or uuid.uuid4().hex)

    def preview(self, preview_id):
        return self.control.get(preview_id, owner=LOCAL_OWNER)

    def get_run(self, preview_id, run_id):
        return self.control.run(preview_id, run_id, owner=LOCAL_OWNER)

    def step(self, preview_id):
        self.control._step(preview_id, self.control._context(preview_id))

    def drive(self, preview_id, predicate, limit=20):
        for _ in range(limit):
            current = self.preview(preview_id)
            if predicate(current):
                return current
            self.step(preview_id)
        self.fail("host coordinator did not reach expected state: " + repr(self.preview(preview_id)))

    def ready(self, *, units=None):
        preview = self.create(units=units or [])
        preview = self.drive(preview["preview_id"], lambda row: row["state"] == "ready")
        return preview, self.service.get_session(preview["session_id"])

    def active(self, *, units=None):
        preview, session = self.ready(units=units)
        run = self.submit(preview["preview_id"], key="A")
        self.drive(preview["preview_id"], lambda row: row["active_run_id"] == run["run_id"])
        return self.preview(preview["preview_id"]), session, run

    def assert_error(self, callback, code=None, status=None):
        with self.assertRaises(ServiceError) as caught:
            callback()
        if code:
            self.assertEqual(caught.exception.code, code)
        if status:
            self.assertEqual(caught.exception.status, status)
        return caught.exception

    def test_create_is_durable_offline_and_freezes_reference_before_allocation(self):
        preview = self.create()
        reopened = PreviewStore(self.control.store.path)
        self.assertEqual(reopened.get_preview(LOCAL_OWNER, preview["preview_id"])["allocation_state"], "pending")
        self.assertEqual(preview["session_request"]["reference_sha256"], self.service.reference_sha)
        self.assertEqual(preview["capabilities"]["application_ready"], "unknown")
        self.assertFalse(self.service.allocations)
        self.assertEqual(self.create(), preview)

    def test_explicit_base_pair_is_frozen_and_replay_never_reselects_it(self):
        for payload in ({"base_firmware_id": "b" * 64}, {"expected_base_sha256": "b" * 64}):
            self.assert_error(lambda: self.create(key=uuid.uuid4().hex, **payload), "uiflow_base_binding_required", 400)
        preview = self.create(base_firmware_id="b" * 64, expected_base_sha256="b" * 64)
        pid = preview["preview_id"]
        self.assertEqual(preview["session_request"]["reference_sha256"], "b" * 64)
        self.assertEqual(preview["request"]["base_firmware_id"], "b" * 64)
        self.assertFalse(preview["capabilities"]["main_mpy"])
        self.assertEqual(self.create(base_firmware_id="b" * 64, expected_base_sha256="b" * 64), preview)
        self.assertEqual(len(self.service.base_calls), 1)
        self.assert_error(lambda: self.create(base_firmware_id="e" * 64, expected_base_sha256="e" * 64), "idempotency_conflict", 409)
        self.control._allocate(self.preview(pid), self.control._context(pid))
        self.assertEqual(self.service.allocations[0][0]["base_firmware_id"], "b" * 64)
        self.assertEqual(self.service.allocations[0][1]["expected_reference_sha256"], "b" * 64)

    def binary_package(self):
        return self.service.program_packages.create({"entrypoint": "main.mpy", "files": {
            "main.mpy": {"encoding": "base64", "data": base64.b64encode(HELPER).decode("ascii")}}})["revision_id"]

    def admit_binary_loader(self, session):
        session._program_capabilities = {**verified_core(), "identity": session._program_identity()}

    def test_binary_pending_or_booting_rejection_has_no_allocation_or_run_intent(self):
        preview = self.create()
        pid, revision = preview["preview_id"], self.binary_package()
        for boot in (False, True):
            if boot:
                self.control._allocate(self.preview(pid), self.control._context(pid))
                session = self.service.get_session(self.preview(pid)["session_id"])
                self.admit_binary_loader(session)
                session.state = "booting"
            before = self.preview(pid)
            count = len(self.service.allocations)
            with mock.patch.object(self.control.store, "admit_run", side_effect=AssertionError("MPY preflight must not admit")):
                self.assert_error(lambda: self.submit(pid, revision, key="binary-before-ready"), "program_mpy_entrypoint_unsupported", 409)
            self.assertEqual(self.preview(pid), before)
            self.assertEqual(len(self.service.allocations), count)

    def test_binary_admission_and_public_capability_use_actual_current_session_only(self):
        created = self.create()
        pid = created["preview_id"]
        self.control._allocate(self.preview(pid), self.control._context(pid))
        session = self.service.get_session(self.preview(pid)["session_id"])
        self.admit_binary_loader(session)
        current = self.preview(pid)
        self.assertTrue(current["capabilities"]["main_mpy"])
        self.assertFalse(current["session_request"]["preview_capabilities"]["main_mpy"])
        self.assertEqual(self.create(), created, "create receipt must not become a canary GET")
        revision = self.binary_package()
        receipt = self.submit(pid, revision, key="verified-binary")
        self.assertEqual(receipt["revision_id"], revision)
        self.assertEqual(receipt["state"], "queued")
        self.control._mutate(pid, lambda state: state["preview"].update(runtime_capabilities=deepcopy(session._program_capabilities)))
        session.pid += 1
        self.assertFalse(self.preview(pid)["capabilities"]["main_mpy"], "persisted proof must not survive process identity change")
        before = self.preview(pid)
        self.assert_error(lambda: self.submit(pid, revision, key="wrong-process"), "program_mpy_entrypoint_unsupported", 409)
        self.assertEqual(self.preview(pid), before)
        session._program_capabilities = {}
        recovered = PreviewService(self.service, _start_workers=False)
        self.controls.append(recovered)
        self.assertFalse(recovered.get(pid, owner=LOCAL_OWNER)["capabilities"]["main_mpy"])
        self.assertFalse(session.run_calls, "recovery must not run a canary or replay a program")

    def test_run_idempotency_replay_precedes_missing_package_and_generation_check(self):
        preview = self.create()
        revision = self.package()
        payload = {"revision_id": revision, "expected_generation": 0, "expected_owner_epoch": preview["owner_epoch"], "timeout_seconds": 120}
        first = self.control.submit(preview["preview_id"], payload, owner=LOCAL_OWNER, idempotency_key="run-A")
        with mock.patch.object(self.service.program_packages, "get", side_effect=AssertionError("replay must not read package")), \
             mock.patch.object(self.service.program_packages, "read_files", side_effect=AssertionError("replay must not read bytes")):
            replay = self.control.submit(preview["preview_id"], deepcopy(payload), owner=LOCAL_OWNER, idempotency_key="run-A")
        self.assertEqual(first, replay)
        self.assertEqual(self.preview(preview["preview_id"])["generation"], 1)
        self.assertFalse(self.service.allocations)

    def test_pending_latest_wins_preserves_durable_run_records(self):

        for ttl in (True, 29, 3601, 3.5):
            self.assert_error(lambda: self.create(key=uuid.uuid4().hex, ttl_seconds=ttl), "invalid_preview_request", 400)

    def test_receipt_reopens_before_transport_and_run_uses_actual_current_owner(self):
        preview, session = self.ready()
        pid = preview["preview_id"]
        receipt = self.submit(pid, key="A")
        def transport(call):
            fresh = PreviewStore(self.control.store.path)
            stored = fresh.get_run(LOCAL_OWNER, pid, receipt["run_id"])
            self.assertEqual(stored["execution_id"], call["execution_id"])
            self.assertEqual(stored["revision_id"], call["revision_id"])
            self.assertIn(stored["state"], {"preparing", "starting"})
            self.assertEqual(fresh.get_preview(LOCAL_OWNER, pid)["transition_run_id"], receipt["run_id"])
            self.assertIsNone(call["expected_execution_id"])
            return {"state": "running", "acknowledged": True}
        session.run_hook = transport
        self.drive(pid, lambda row: row["active_run_id"] == receipt["run_id"])
        self.assertEqual(len(session.run_calls), 1)
        self.assertEqual(self.get_run(pid, receipt["run_id"])["application_ready"], "unknown")

    def test_known_bad_package_does_not_cancel_active_a_or_close_input_admission(self):
        preview, session, old = self.active()
        pid = preview["preview_id"]
        context = self.control._context(pid)
        token = context.cancel_event
        revision = self.package("invalid", **{"native.mpy": {"encoding": "base64", "data": "TQYBHw=="}})
        self.assert_error(lambda: self.submit(pid, revision, key="bad"), "program_mpy_native_unsupported")
        self.assertEqual(self.preview(pid)["active_run_id"], old["run_id"])
        self.assertEqual(self.preview(pid)["generation"], old["generation"])
        self.assertEqual(self.get_run(pid, old["run_id"])["state"], "running")
        self.assertTrue(context.admission)
        if token is not None:
            self.assertFalse(token.is_set())
        self.assertEqual(len(session.run_calls), 1)
        self.assertFalse(session.interrupt_calls)

    def test_binary_checkpoint_does_not_change_active_run_or_cancel_input_admission(self):
        preview, session, old = self.active()
        pid = preview["preview_id"]
        context = self.control._context(pid)
        token = context.cancel_event
        revision = self.service.program_packages.create({
            "entrypoint": "main.mpy",
            "files": {"main.mpy": {"encoding": "base64", "data": base64.b64encode(HELPER).decode("ascii")}},
        })
        before = self.preview(pid)
        old_record = self.get_run(pid, old["run_id"])
        with mock.patch.object(self.control.store, "admit_run", side_effect=AssertionError("unsupported entrypoint must not allocate a Run")):

        self.assertEqual(self.preview(pid), before)
        self.assertEqual(self.get_run(pid, old["run_id"]), old_record)
        self.assertTrue(context.admission)
        if token is not None:
            self.assertFalse(token.is_set())
        self.assertEqual(len(session.run_calls), 1)
        self.assertFalse(session.interrupt_calls)

    def test_cancelled_transfer_late_success_waits_for_independent_prompt_before_c(self):
        preview, session, old = self.active()
        pid = preview["preview_id"]
        transfer, interrupt = self.block(), self.block()
        def late_success(call):
            transfer.wait()
            self.assertTrue(call["cancelled"]())
            return {"state": "running", "acknowledged": True}
        session.run_hook = late_success
        middle = self.submit(pid, key="B")
        thread, outcome = self.background(lambda: self.drive(pid, lambda row: len(session.run_calls) == 2))
        self.assertTrue(transfer.entered.wait(1))
        newest = self.submit(pid, key="C")
        self.assertEqual(self.get_run(pid, middle["run_id"])["state"], "cancelling")
        transfer.release.set()
        thread.join(timeout=2)
        self.assertNotIn("error", outcome)
        self.assertNotEqual(self.preview(pid)["active_run_id"], middle["run_id"])
        self.assertNotIn(self.get_run(pid, middle["run_id"])["state"], {"running", "completed"})
        owner = session._owner()
        self.assertEqual(owner, middle["execution_id"], "transport owner differs from active pointer")
        def prompt():
            interrupt.wait()
            return {"interruption_confirmed": True, "prompt_end": 50, "serial_offset": 30}
        session.interrupt_hook = prompt
        recovery, result = self.background(lambda: self.step(pid))
        self.assertTrue(interrupt.entered.wait(1))
        self.assertEqual(session.interrupt_calls[-1]["expected_execution_id"], owner)
        self.assertEqual(len(session.run_calls), 2, "C cannot start before prompt acknowledgement")
        self.assertNotEqual(self.get_run(pid, middle["run_id"])["state"], "cancelled")

    def test_concurrent_true_false_button_backend_order_cannot_lose_final_release(self):
        preview, session, old = self.active(units=[{"id": "fixture-button", "unit_id": "button", "port_id": "port-a"}])
        pid = preview["preview_id"]
        pressed, block, false_done = {"value": False}, self.block(), threading.Event()
        physical = []
        def button(command):
            value = command["values"]["pressed"]
            if value:
                block.wait()
            pressed["value"] = value
            physical.append(value)
            if not value:
                false_done.set()
            return {"accepted": True}
        session.unit_input = button
        def payload(value):
            return {"expected_owner_epoch": preview["owner_epoch"], "expected_run_id": old["run_id"],

        self.assertTrue(block.entered.wait(1))

        # Old concurrent code can finish False before the delayed True. New
        # serialized code may hold False until True completes: both schedules
        # must finish safely, so never require False-first to release True.
        false_done.wait(.05)
        block.release.set()

        self.drive(pid, lambda row: self.get_run(pid, old["run_id"])["state"] == "cancelled")
        self.assertTrue(physical)
        self.assertFalse(physical[-1], "last actual backend effect must release the momentary input")

    def test_false_then_queued_true_with_partial_backend_failure_preserves_release_obligation(self):
        preview, session, old = self.active(units=[{"id": "fixture-button", "unit_id": "button", "port_id": "port-a"}])
        pid = preview["preview_id"]
        physical, state, block = [], {"pressed": False, "first_false": True}, self.block()
        def button(command):
            value = command["values"]["pressed"]
            if not value and state["first_false"]:
                block.wait()
                state["first_false"] = False
            state["pressed"] = value
            physical.append(value)
            if value:
                raise ServiceError("qmp_unconfirmed", "HOST_ONLY_NOT_GUEST partial True effect", 504)
            return {"accepted": True}
        session.unit_input = button
        def payload(value):
            return {"expected_owner_epoch": preview["owner_epoch"], "expected_run_id": old["run_id"],
                    "unit_input": {"id": "fixture-button", "values": {"pressed": value}}}
        up, up_result = self.background(lambda: self.control.input(pid, payload(False), owner=LOCAL_OWNER))
        self.assertTrue(block.entered.wait(1))
        down, down_result = self.background(lambda: self.control.input(pid, payload(True), owner=LOCAL_OWNER))
        deadline = time.monotonic() + 1
        while self.control._context(pid).inputs < 2 and time.monotonic() < deadline:
            time.sleep(.005)
        self.assertEqual(self.control._context(pid).inputs, 2, "True must be admitted and queued behind the blocked False")
        block.release.set()
        up.join(timeout=2)
        down.join(timeout=2)
        self.assertNotIn("error", up_result)
        self.assertIsInstance(down_result.get("error"), ServiceError)
        self.assertEqual(down_result["error"].code, "qmp_unconfirmed")
        self.assertEqual(physical, [False, True])
        self.assertTrue(state["pressed"])
        self.assertTrue(self.preview(pid)["held_inputs"], "failed True may have an effect and still needs a release")
        self.control.cancel(pid, old["run_id"], {"expected_owner_epoch": preview["owner_epoch"]}, owner=LOCAL_OWNER)
        self.drive(pid, lambda row: self.get_run(pid, old["run_id"])["state"] == "cancelled")
        self.assertFalse(physical[-1])
        self.assertFalse(state["pressed"])
        self.assertFalse(self.preview(pid)["held_inputs"])

    def test_preview_input_duration_above_five_seconds_is_rejected_before_backend_or_held_write(self):
        preview, session, old = self.active()
        pid = preview["preview_id"]
        common = {"expected_owner_epoch": preview["owner_epoch"], "expected_run_id": old["run_id"]}
        for command in (
            {"kind": "key", "key": "a", "action": "down", "settle_seconds": 5.01},
            {"kind": "longpress", "x": 2, "y": 2, "hold_seconds": 5.01},
        ):
            self.assert_error(lambda: self.control.input(pid, {**common, "input": command}, owner=LOCAL_OWNER), status=400)
            self.assertFalse(self.preview(pid).get("held_inputs"))
            self.assertFalse(session.inputs)
            self.assertEqual(self.control._context(pid).inputs, 0)
