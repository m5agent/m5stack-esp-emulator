
def png_pixels(data, positions):
 assert data.startswith(b'\x89PNG\r\n\x1a\n')
 offset=8;header=None;compressed=bytearray()
 while offset<len(data):
  size=struct.unpack_from('>I',data,offset)[0];kind=data[offset+4:offset+8];body=data[offset+8:offset+8+size]
  if kind==b'IHDR':header=struct.unpack('>IIBBBBB',body)
  elif kind==b'IDAT':compressed.extend(body)
  offset+=size+12
 width,height,bits,color,compression,filtering,interlace=header
 assert bits==8 and color in (2,6) and compression==filtering==interlace==0
 channels=3 if color==2 else 4;stride=width*channels;raw=zlib.decompress(compressed)
 assert len(raw)==height*(stride+1)
 previous=bytearray(stride);result={}
 for y in range(height):
  at=y*(stride+1);mode=raw[at];row=bytearray(raw[at+1:at+1+stride]);assert 0<=mode<=4
  for x in range(stride):
   left=row[x-channels] if x>=channels else 0;up=previous[x];upper=previous[x-channels] if x>=channels else 0
   if mode==1:value=left
   elif mode==2:value=up
   elif mode==3:value=(left+up)//2
   elif mode==4:
    p=left+up-upper;a,b,c=abs(p-left),abs(p-up),abs(p-upper);value=left if a<=b and a<=c else up if b<=c else upper
   else:value=0
   row[x]=(row[x]+value)&255
  for name,(px,py) in positions.items():
   assert 0<=px<width and 0<=py<height
   if y==py:result[name]=list(row[px*channels:px*channels+3])
  previous=row
 assert set(result)==set(positions)
 return {'size':[width,height],'pixels':result}
