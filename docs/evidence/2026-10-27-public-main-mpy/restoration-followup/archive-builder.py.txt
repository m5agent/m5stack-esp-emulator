def clean_text(value):
 value=value.replace(str(ROOT),'$REPO').replace(str(Path.home()),'$HOME')
 value=mac.sub('[redacted-mac]',value)
 def address(m):
  try:ip=ipaddress.ip_address(m.group())
  except ValueError:return m.group()
  if ip.is_private and not ip.is_loopback and not ip.is_unspecified:return '[redacted-private-ip]'
  return m.group()
 return private_ip.sub(address,value)

def sanitize(value,changes,path='$'):
 if isinstance(value,list):return [sanitize(item,changes,path+f'[{i}]') for i,item in enumerate(value)]
 if isinstance(value,str):
  result=clean_text(value)
  if result!=value:changes.append({'path':path,'kind':'local-path-or-address-redaction'})
  return result
 if not isinstance(value,dict):return value
 # Every public package format stores encoded file data in this shape.
 if value.get('encoding')=='base64' and isinstance(value.get('data'),str):
  try:raw=base64.b64decode(value['data'],validate=True)
  except ValueError:raw=None
  if raw is not None:
   changes.append({'path':path,'kind':'base64-payload-omitted','decoded_bytes':len(raw),'decoded_sha256':sha(raw)})
   return {'archive_redacted_binary':True,'original_encoding':'base64','decoded_bytes':len(raw),'decoded_sha256':sha(raw)}
 result={}
 for key,item in value.items():
  if key.lower() in secret_keys and item not in (None,'',False,[]):
   result[key]='[redacted-secret-field]';changes.append({'path':path+'.'+key,'kind':'secret-field-redaction'})
  else:result[key]=sanitize(item,changes,path+'.'+key)
 return result

def register(origin,raw,rel,*,force_hash=None):
 original={'original_source':origin,'original_bytes':len(raw),'original_sha256':sha(raw)}
 if force_hash or raw.startswith(b'PK\x03\x04') or Path(rel).suffix.lower() in {'.zip','.mpy','.bin','.rom','.elf'}:
  original.update(classification='hash-only',archive_path=None,reason=force_hash or 'binary-container-or-program-policy')
  rows.append(original);return original
 changes=[]
 try:value=json.loads(raw)
 except (ValueError,UnicodeError):value=None
 if value is not None:
  changed=sanitize(value,changes)
  if changes:archived=encode({'archive_notice':{'classification':'redacted-json','original_source':origin,'original_sha256':sha(raw),'original_bytes':len(raw),'transformations':changes},'document':changed})
  else:archived=raw
 elif raw.startswith(b'\x89PNG\r\n\x1a\n'):
  archived=raw
 else:
  try:text=raw.decode('utf-8')
  except UnicodeError:
   original.update(classification='hash-only',archive_path=None,reason='non-UTF8-non-PNG-binary');rows.append(original);return original
  changed=clean_text(text)
  if changed!=text:
   changes=[{'kind':'local-path-or-address-redaction'}]
   archived=(f'ARCHIVE REDACTED TEXT; original SHA256 {sha(raw)}; original bytes {len(raw)}\n'+changed).encode()
  else:archived=raw
 rel=rel+'.txt' if rel.endswith('.md') else rel
 if changes:rel += '.redacted.json' if value is not None else '.redacted.txt'
 dest=OUT/rel;dest.parent.mkdir(parents=True,exist_ok=True);dest.write_bytes(archived)
 original.update(classification=('redacted-json' if value is not None else 'redacted-text') if changes else 'original',archive_path=rel,archive_bytes=len(archived),archive_sha256=sha(archived),transformations=changes)
 rows.append(original);return original

def generated(rel,raw):
 dest=OUT/rel;dest.parent.mkdir(parents=True,exist_ok=True);dest.write_bytes(raw)
 rows.append({'classification':'generated','original_source':None,'original_bytes':None,'original_sha256':None,'archive_path':rel,'archive_bytes':len(raw),'archive_sha256':sha(raw)})
